def map_cluster_to_emotion(cluster_id: int, 
                          labeled_tracks: Dict[int, EmotionVector]) -> EmotionVector:
    """Find emotion centroid for each cluster"""
    cluster_tracks = [t for t, c in zip(track_ids, cluster_labels) if c == cluster_id]
    labeled_cluster_tracks = [t for t in cluster_tracks if t in labeled_tracks]
    
    if len(labeled_cluster_tracks) < 10:  # Minimum threshold
