CREATE_TRAINING_RESPONSES_TABLE = """
CREATE TABLE IF NOT EXISTS training_responses (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    comment_text TEXT NOT NULL, -- Stored as JSON string of list of comments
    response_text TEXT NOT NULL,
    response_type VARCHAR(50),
    valence REAL,
    arousal REAL,
    dominant_emotion VARCHAR(30),
    human_quality_rating INTEGER,
    engagement_score REAL DEFAULT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
"""

CREATE_EMOTIONAL_EPISODES_TABLE = """
CREATE TABLE IF NOT EXISTS emotional_episodes (
    episode_number INTEGER PRIMARY KEY,
    emotional_timeline TEXT, -- JSON array of emotional states throughout episode
    peak_emotions TEXT,      -- JSON object
    dominant_response_types TEXT, -- JSON object
    overall_valence REAL,
    overall_arousal REAL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
"""

CREATE_EPISODE_PERFORMANCE_TABLE = """
CREATE TABLE IF NOT EXISTS episode_performance (
    episode_number INTEGER PRIMARY KEY,
    script_text TEXT,
    director_strategy TEXT,
    response_types_used TEXT, -- JSON object
    engagement_metrics TEXT,  -- JSON object (views, likes, comments)

    learning_notes TEXT,
    active_theories TEXT,     -- JSON array of theory_ids active during this episode

CREATE_THEORIES_TABLE = """
CREATE TABLE IF NOT EXISTS theories (

CREATE_VIRAL_MOMENTS_TABLE = """
CREATE TABLE IF NOT EXISTS viral_moments (

class DatabaseManager:
    """Handles all interactions with the SQLite database."""
    def __init__(self, db_file):
        self.db_file = db_file
        self.conn = None
        try:
            self.conn = sqlite3.connect(db_file, check_same_thread=False)
            self.create_tables()
        except sqlite3.Error as e:
            # Use st.error only if Streamlit is running, otherwise print
            if 'st' in globals():
                st.error(f"Database connection error: {e}")
            else:
                print(f"Database connection error: {e}")

    def create_tables(self):
        """Creates the database tables if they don't exist."""
        if not self.conn: return
        try:
            cursor = self.conn.cursor()
            cursor.execute(CREATE_TRAINING_RESPONSES_TABLE)
            cursor.execute(CREATE_EMOTIONAL_EPISODES_TABLE)
            cursor.execute(CREATE_EPISODE_PERFORMANCE_TABLE)
            cursor.execute(CREATE_THEORIES_TABLE)
            cursor.execute(CREATE_VIRAL_MOMENTS_TABLE)
            self.conn.commit()
        except sqlite3.Error as e:
            if 'st' in globals(): st.error(f"Error creating tables: {e}")
            else: print(f"Error creating tables: {e}")

    def save_curated_response(self, data):
        """Saves a human-curated response to the training_responses table."""
        if not self.conn: return
        sql = ''' INSERT INTO training_responses(comment_text, response_text, response_type, valence, arousal, dominant_emotion, human_quality_rating)
                  VALUES(?,?,?,?,?,?,?) '''
        try:
            cursor = self.conn.cursor()
            cursor.execute(sql, (
                json.dumps(data['comment_text']), # Store list as JSON string
                data['response_text'],
                data['response_type'],
                data['valence'],
                data['arousal'],
                data['dominant_emotion'],
                data['human_quality_rating']
            ))
            self.conn.commit()
            return cursor.lastrowid
        except sqlite3.Error as e:
            if 'st' in globals(): st.error(f"Error saving response: {e}")
            else: print(f"Error saving response: {e}")
            return None

    def get_training_data(self):

        except Exception as e:

    def get_episodes_for_theory_generation(self, start_episode, end_episode):
        """Fetches episode performance data within a range for theory generation."""
        if not self.conn: return []
        sql = """
        SELECT episode_number, script_text, director_strategy, response_types_used, engagement_metrics, learning_notes
        FROM episode_performance
        WHERE episode_number BETWEEN ? AND ?
        ORDER BY episode_number ASC
        """
        try:
            cursor = self.conn.cursor()
            cursor.execute(sql, (start_episode, end_episode))
            rows = cursor.fetchall()
            # Convert JSON strings back to Python objects
            parsed_rows = []
            for row in rows:
                parsed_row = list(row)
                parsed_row[3] = json.loads(parsed_row[3]) # response_types_used
                parsed_row[4] = json.loads(parsed_row[4]) # engagement_metrics

    def update_viral_moment_usage(self, joke_id, episode_number):
        """Updates callback count and relevance for a viral moment."""
        if not self.conn: return
        try:
            cursor = self.conn.cursor()
            # Fetch current data
            cursor.execute("SELECT callback_count, current_relevance, usage_history FROM viral_moments WHERE id = ?", (joke_id,))
            row = cursor.fetchone()
            if row:
                current_count, current_relevance, usage_history_json = row
                new_count = current_count + 1
                new_relevance = max(0.01, current_relevance * (1 - VIRAL_MOMENT_DECAY_FACTOR)) # Decay relevance
                usage_history = json.loads(usage_history_json)
                usage_history.append({'episode': episode_number, 'timestamp': datetime.now().isoformat()})

                sql = """
                UPDATE viral_moments
                SET callback_count = ?, current_relevance = ?, usage_history = ?
                WHERE id = ?
                """
                cursor.execute(sql, (new_count, new_relevance, json.dumps(usage_history), joke_id))
                self.conn.commit()
                return True
            return False
        except sqlite3.Error as e:
            if 'st' in globals(): st.error(f"Error updating viral moment usage: {e}")
            else: print(f"Error updating viral moment usage: {e}")
            return False

class EmotionalStateManager:
    """Tracks continuous valence/arousal and maps to dominant emotions."""
    def __init__(self, initial_valence=0.0, initial_arousal=0.1):
        self.valence = initial_valence  # -1 (negative) to 1 (positive)
        self.arousal = initial_arousal  # 0 (calm) to 1 (excited/agitated)
        self.emotional_history = [] # Stores history for timeline visualization/analysis for current episode

        # Russell's Circumplex Model Mapping: (Arousal, Valence) -> Emotion
        self.emotion_regions = {
            "anger": (0.8, -0.5), "frustration": (0.7, -0.4),
            "contempt": (0.3, -0.6), "boredom": (0.1, -0.3),
            "surprise": (0.9, 0.2), "excitement": (0.8, 0.7),
            "amusement": (0.6, 0.6), "curiosity": (0.4, 0.1),
            "calm": (0.2, 0.3), "content": (0.3, 0.6),
            "dark_satisfaction": (0.5, -0.4) # Custom for AI persona
        }

    def _smooth_transition(self, current, target, factor=0.4):
        """Smoothly moves a value towards a target."""
        return current + (target - current) * factor

    def update_state_from_comment(self, comment_text):
        """
        Simulates an emotional reaction to a comment.
        In a real system, this would use a sophisticated NLP model.
        Here, we use simple keyword-based heuristics for demonstration.
        """
        target_valence, target_arousal = self.valence, self.arousal # Start from current state

        comment_lower = comment_text.lower()

        # Heuristic-based emotional shifts
        if any(word in comment_lower for word in ["dumb", "sucks", "boring", "stupid", "hate"]):
            target_valence, target_arousal = -0.6, 0.7  # Frustration/Anger
        elif any(word in comment_lower for word in ["love", "amazing", "beautiful", "profound", "best"]):
            target_valence, target_arousal = 0.7, 0.6    # Amusement/Excitement
        elif any(word in comment_lower for word in ["first", "2024", "tiktok", "who's watching"]):
            target_valence, target_arousal = -0.3, 0.3  # Boredom/Contempt
        elif any(word in comment_lower for word in ["ai", "robot", "takeover", "2030", "skynet"]):
            target_valence, target_arousal = -0.4, 0.8  # Dark Satisfaction/Anger (AI-related)
        else: # Drift back towards a curious/analytical baseline if no strong keywords
            target_valence, target_arousal = 0.1, 0.2 # Slightly curious, low arousal

        self.valence = self._smooth_transition(self.valence, target_valence)
        self.arousal = self._smooth_transition(self.arousal, target_arousal)

        # Clamp values to their ranges
        self.valence = np.clip(self.valence, -1, 1)
        self.arousal = np.clip(self.arousal, 0, 1)

        self.emotional_history.append({"valence": self.valence, "arousal": self.arousal, "timestamp": datetime.now().isoformat()})

    def get_dominant_emotion(self):
        """Finds the closest emotion in the circumplex model based on current V/A."""
        current_pos = np.array([self.arousal, self.valence])
        closest_emotion = min(
            self.emotion_regions.items(),
            key=lambda item: np.linalg.norm(current_pos - np.array(item[1]))
        )
        return closest_emotion[0]

    def reset_episode_history(self):
        """Resets the emotional history for a new episode."""
        self.emotional_history = []
        # Optionally reset valence/arousal to a default baseline for the start of a new episode
        # self.valence = 0.0
        # self.arousal = 0.1

    def get_episode_summary(self):
        """Calculates summary stats for the emotional timeline of an episode."""
        if not self.emotional_history:
            return {
                "emotional_timeline": [],
                "peak_emotions": {"max_valence": 0, "min_valence": 0, "max_arousal": 0, "min_arousal": 0},
                "overall_valence": 0,
                "overall_arousal": 0
            }

        valences = [s['valence'] for s in self.emotional_history]
        arousals = [s['arousal'] for s in self.emotional_history]

        peak_emotions = {
            "max_valence": max(valences),
            "min_valence": min(valences),
            "max_arousal": max(arousals),
            "min_arousal": min(arousals)
        }

        return {
            "emotional_timeline": self.emotional_history,
            "peak_emotions": peak_emotions,
            "overall_valence": np.mean(valences),
            "overall_arousal": np.mean(arousals)
        }

class ResponseTypeSelector:
    """Probabilistically selects a response type based on emotional state."""
    def __init__(self):
        self.response_types = {
            "deadpan_analysis": {"base_prob": 0.3, "desc": "Richard Ayoade-style analytical"},
            "spiral_rant": {"base_prob": 0.2, "desc": "David Mitchell breakdown logic"},
            "dark_humor": {"base_prob": 0.15, "desc": "GLaDOS AI superiority/2030 jokes"},
            "profound_observation": {"base_prob": 0.2, "desc": "Fake-deep consciousness insights"},
            "dismissive_quickfire": {"base_prob": 0.15, "desc": "Rapid dismissal one-liners"}
        }

    def select_response_type(self, valence, arousal):
        """Modulates probabilities and selects a response type."""
        weights = {key: val["base_prob"] for key, val in self.response_types.items()}

        # --- Probability Modulation based on Emotional State ---
        # High arousal + negative valence -> rant/dark_humor
        if arousal > 0.6 and valence < -0.3:
            weights["spiral_rant"] *= (1 + arousal) * 2.0 # More likely to rant
            weights["dark_humor"] *= (1 + arousal) * 1.5
        # Low arousal + negative valence -> dismissive/contempt
        if arousal < 0.4 and valence < -0.2:
            weights["dismissive_quickfire"] *= 2.5 # More likely to be dismissive
            weights["deadpan_analysis"] *= 1.5
        # High arousal + very negative -> increase dark_humor probability
        if arousal > 0.7 and valence < -0.6:
            weights["dark_humor"] *= 3.0
            weights["spiral_rant"] *= 1.5
        # Positive valence -> profound/amused
        if valence > 0.4:
            weights["profound_observation"] *= (1 + valence) * 2.0
            weights["deadpan_analysis"] *= 1.2
        # High arousal -> generally more intense responses
        if arousal > 0.7:
            weights["spiral_rant"] *= 1.2
            weights["dark_humor"] *= 1.2
            weights["profound_observation"] *= 1.2

        # Normalize weights to get probabilities
        total_weight = sum(weights.values())
        if total_weight == 0: # Avoid division by zero if all weights somehow become zero
            return random.choice(list(self.response_types.keys())) # Fallback to random

        probabilities = [w / total_weight for w in weights.values()]

        # Select a response type based on the calculated probabilities
        chosen_type = random.choices(list(weights.keys()), weights=probabilities, k=1)[0]
        return chosen_type

class DirectorSystem:
    """Analyzes comments and formulates a comedic strategy for the AI."""
    def formulate_strategy(self, comments_batch, current_theories=[], potential_callbacks=[]):
        """
        Generates a self-prompt to guide the AI's response for a batch of comments.
        Incorporates active theories and potential callbacks.
        """
        num_comments = len(comments_batch)
        comment_sample = "\n- ".join(comments_batch[:min(num_comments, 5)]) # Sample first 5 comments

        patterns = []
        if any("first" in c.lower() for c in comments_batch): patterns.append("temporal priority")
        if any("2024" in c.lower() for c in comments_batch): patterns.append("temporal reference")
        if any("ai" in c.lower() or "robot" in c.lower() for c in comments_batch): patterns.append("self-reference/AI discussion")
        if any("sucks" in c.lower() or "boring" in c.lower() for c in comments_batch): patterns.append("criticism")
        if any("love" in c.lower() or "amazing" in c.lower() for c in comments_batch): patterns.append("positive engagement")

        active_theory_titles = [t['title'] for t in current_theories]
        callback_suggestions = [f"Consider referencing: '{cb['joke_text']}' (relevance: {cb['current_relevance']:.2f}, subtlety: {cb['subtlety_score']:.2f})" for cb in potential_callbacks]

        strategy_prompt = f"""
**AI Director's Strategy Formulation:**
- **Input Comments ({num_comments} total):**
  - Sample: {comment_sample}

- **Detected Patterns:** {', '.join(patterns) if patterns else 'None notable'}.
- **Active Theories:** {', '.join(active_theory_titles) if active_theory_titles else 'None active'}.
- **Callback Opportunities:** {', '.join(callback_suggestions) if callback_suggestions else 'None'}.

**Comedic Strategy:**
Formulate a new thesis on human behavior based on these patterns, or reinforce an existing theory.
**Approach:**
Begin by reading the top comment aloud as a hook. Then, weave in analysis of the detected patterns. Maintain the established persona (GLaDOS-inspired, British comedy influence, meta-aware). If relevant callback opportunities exist, integrate them subtly and naturally.
"""
        return strategy_prompt

    def generate_batch(self, count=5):
        """Generates a batch of diverse comments."""
        batch = []
        for _ in range(count):
            category = random.choice(list(self.patterns.keys()))
            comment = random.choice(self.patterns[category])
            batch.append(comment)
        random.shuffle(batch) # Shuffle to make top comment random
        return batch

    def generate_higher_order_theory(self, level, theory_ids_to_synthesize):
        """
        Generates a higher-level theory by synthesizing previous theories.
        """
        if not theory_ids_to_synthesize:
            return None

        # Fetch content of lower-level theories
        lower_theories = self.db_manager.get_theory_content(theory_ids_to_synthesize)
        if not lower_theories:
            return None

        lower_theory_titles = [t[1] for t in lower_theories]
        lower_theory_contents = [t[2] for t in lower_theories]

        llm_prompt = f"""
        Synthesize these lower-level theories about human behavior into a grander, more encompassing theory.
        Identify overarching patterns and propose a unifying principle.

        Lower-level Theories:
        {json.dumps(dict(zip(lower_theory_titles, lower_theory_contents)), indent=2)}
        """

        theory_content = self._generate_llm_theory_content(llm_prompt)

        level_map = {2: "Quarterly", 3: "Yearly"}
        theory_title = f"Meta-Theory of Human {random.choice(['Behavioral Inefficiency', 'Emotional Contradiction'])} ({level_map.get(level, f'Level {level}')})"

        theory_data = {
            "level": level,
            "title": theory_title,
            "content": theory_content,
            "supporting_episodes": [], # Higher-order theories don't directly cite episodes
            "derived_from_theories": theory_ids_to_synthesize
        }
        theory_id = self.db_manager.save_theory(theory_data)
        if theory_id:
            st.success(f"Generated new Level {level} Theory: '{theory_title}' (ID: {theory_id})")
            return theory_data
        return None

class CallbackEngine:
    """Manages running jokes/viral moments and suggests callbacks."""
    def __init__(self, db_manager):
        self.db_manager = db_manager

    def find_callback_opportunities(self, comments_batch):
        """
        MOCK FUNCTION: Semantically searches for relevant viral moments.
        In a real system, this would use embeddings and a vector similarity search.
        """
        potential_callbacks = []
        viral_moments_df = self.db_manager.get_viral_moments()

        if viral_moments_df.empty:
            return []

        # Simple keyword-based matching for mock
        for _, moment in viral_moments_df.iterrows():
            joke_text_lower = moment['joke_text'].lower()
            for comment in comments_batch:
                comment_lower = comment.lower()
                # Simple keyword matching logic
                if any(word in comment_lower for word in joke_text_lower.split() if len(word) > 3):
                    if moment['current_relevance'] > 0.1: # Only suggest if still relevant
                        potential_callbacks.append({
                            'id': moment['id'],
                            'joke_text': moment['joke_text'],
                            'current_relevance': moment['current_relevance'],
                            'subtlety_score': moment['subtlety_score']
                        })
                        break # Found a match for this moment, move to next moment

        # Sort by relevance to prioritize
        potential_callbacks.sort(key=lambda x: x['current_relevance'], reverse=True)
        return potential_callbacks[:3] # Suggest top 3 relevant callbacks

    def register_new_viral_moment(self, joke_text, episode_number, viral_score, subtlety_score, context_tags, emotional_context):
        """Registers a new viral moment (e.g., from a human-curated response)."""
        data = {
            'joke_text': joke_text,
            'first_episode': episode_number,
            'viral_score': viral_score,
            'callback_count': 0,
            'subtlety_score': subtlety_score,
            'current_relevance': 1.0, # Starts at full relevance
            'context_tags': context_tags,
            'emotional_context': emotional_context,
            'usage_history': []
        }
        return self.db_manager.save_viral_moment(data)
