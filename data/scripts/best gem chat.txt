from sqlalchemy import Column, Integer, String, Text, Float, DateTime, Boolean, JSON

from sqlalchemy.ext.declarative import declarative_base

from datetime import datetime

Base = declarative_base()

class Episode(Base):

    tablename = "episodes"

    id = Column(Integer, primary_key=True)

    episode_number = Column(Integer, unique=True, nullable=False)

    top_comment = Column(Text, nullable=False)

    # MVP: Only store the top comment for simplicity

    ai_response = Column(Text)

    analysis_phase = Column(Text)

    reflection_phase = Column(Text)

    audio_file_path = Column(String(255))

    generation_time = Column(Float)

    # MVP: Simplified quality score

    overall_quality_score = Column(Float, default=0.0)

    # MVP: Simplified persona snapshot to core traits

    persona_snapshot = Column(JSON, default={}) 

    created_at = Column(DateTime, default=datetime.utcnow)

    updated_at = Column(DateTime, default=datetime.utcnow, onupdate=datetime.utcnow)

class PersonaState(Base): # MVP: Simplified to core traits only

    tablename = "persona_states"

    id = Column(Integer, primary_key=True)

    episode_number = Column(Integer, nullable=False, unique=True)

    analytical_level = Column(Float, default=0.0)

    confusion_level = Column(Float, default=0.0)

    meta_awareness = Column(Float, default=0.0)

    humor_sophistication = Column(Float, default=0.0)

    # MVP: No emotional_dimensions, motivational_drives, dialogue_style_preference, strategy_preference

    created_at = Column(DateTime, default=datetime.utcnow)

class Theory(Base): # MVP: Simplified, no direct relationship to episodes yet

    tablename = "theories"

    id = Column(Integer, primary_key=True)

    name = Column(String(200), nullable=False, unique=True)

    description = Column(Text)

    confidence_level = Column(Float, default=0.5)

    category = Column(String(50))

    humor_angle = Column(Text)

    confusion_point = Column(Text)

    evidence_strength = Column(Float, default=0.5)

    # MVP: No direct episode links or refinement counts

    created_at = Column(DateTime, default=datetime.utcnow)

    updated_at = Column(DateTime, default=datetime.utcnow, onupdate=datetime.utcnow)
