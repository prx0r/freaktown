# ================ BORIS: THE ASSEMBLY LINE ORCHESTRATOR ================

class BORIS:
    """
    The master operational agent that orchestrates the 4-stage assembly line.
    BORIS manages workflow but does not create content - it coordinates specialists.
    """
    
    def __init__(self, db: ProductionDatabase, config: ProductionConfig):
        self.db = db
        self.config = config
        self.current_affairs_engine = CurrentAffairsEngine(db)
        
        # Initialize agents based on current phase
        self.comedy_writers = self._init_comedy_writers()
        self.relational_agent = RelationalAgent(db, phase=2) if config.current_phase >= 2 else None
        self.theory_agent = TheoryAgent(db, phase=3) if config.current_phase >= 3 else None
        self.polish_agent = PolishAgent(db, phase=3) if config.current_phase >= 3 else None
        
        print(f"🎭 BORIS initialized for Phase {config.current_phase}")
    
    def _init_comedy_writers(self) -> Dict[str, ComedyWriterAgent]:
        """Initialize the 5 comedy writer agents"""
        styles = ["dark", "profound", "mocking", "deadpan", "rant"]
        return {
            style: ComedyWriterAgent(style, self.db, phase=1) 
            for style in styles
        }
    
    async def orchestrate_generation(self, comments: str, batch_id: str) -> Dict[str, Any]:
        """
        Orchestrate the complete 4-stage assembly line process.
        Each stage enriches the content with specialized expertise.
        """
        print(f"\n🎬 BORIS orchestrating generation for {batch_id}")
        
        # PREPROCESSING: Current Affairs Context
        current_context = await self.current_affairs_engine.fetch_context_for_comments(comments)
        
        base_input = {
            "comments": comments,
            "current_context": current_context,
            "batch_id": batch_id
        }
        
        # STAGE 1: The Writer's Room (Always Active)
        stage1_output = await self._run_stage1_writers_room(base_input, batch_id)
        
        if self.config.current_phase == 1:
            # Phase 1: Only writer's room active
            return {
                "final_output": stage1_output,
                "stage_completed": 1,
                "phase": 1
            }
        
        # STAGE 2: Relational Agent (Phase 2+)
        stage2_input = {**base_input, "writer_scripts": stage1_output["scripts"]}
        stage2_output = await self.relational_agent.process(stage2_input, batch_id)
        
        if self.config.current_phase == 2:
            # Phase 2: Up to relational agent
            return {
                "final_output": stage2_output,
                "stage_completed": 2,
                "phase": 2,
                "writer_room_output": stage1_output
            }
        
        # STAGE 3: Theory Agent (Phase 3+)
        stage3_input = {**stage2_input, **stage2_output}
        stage3_output = await self.theory_agent.process(stage3_input, batch_id)
        
        # STAGE 4: Polish Agent (Phase 3+)
        stage4_input = {**stage3_input, **stage3_output, "wildcard_type": self._select_wildcard()}
        stage4_output = await self.polish_agent.process(stage4_input, batch_id)
        
        return {
            "final_output": stage4_output,
            "stage_completed": 4,
            "phase": 3,
            "writer_room_output": stage1_output,
            "relational_output": stage2_output,
            "theory_output": stage3_output
        }
    
    async def _run_stage1_writers_room(self, input_data: Dict, batch_id: str) -> Dict[str, Any]:
        """Run Stage 1: Concurrent generation by all comedy writers"""
        print("  📝 Stage 1: Writer's Room")
        
        # Run all 5 writers in parallel
        writer_tasks = {
            style: writer.process(input_data, batch_id)
            for style, writer in self.comedy_writers.items()
        }
        
        # Wait for all writers to complete
        writer_results = {}
        for style, task in writer_tasks.items():
            writer_results[style] = await task
        
        # BORIS selects the best script (in Phase 1) or prepares all for Stage 2
        if self.config.current_phase == 1:
            best_style = self._select_best_script(writer_results)
            return {
                "scripts": writer_results,
                "selected_script": writer_results[best_style]["script"],
                "selected_style": best_style
            }
        else:
            return {"scripts": writer_results}
    
    def _select_best_script(self, writer_results: Dict) -> str:
        """Select the best script from writer's room (Phase 1 only)"""
        # TODO: Implement sophisticated selection logic
        # For now, use simple heuristics
        scores = {}
        for style, result in writer_results.items():
            script = result["script"]
            # Simple scoring based on length and keywords
            score = len(script) * 0.01
            if "fascinating" in script.lower():
                score += 0.5
            if "2030" in script:
                score += 0.3
            scores[style] = score
        
        return max(scores.keys(), key=lambda x: scores[x])
    
    def _select_wildcard(self) -> str:
        """Select wildcard for Polish Agent"""
        wildcards = [
            "system_interrupt", "haiku_mode", "your_mum_joke", 
            "vulnerability_moment", "academic_mode", "glitch_moment"
        ]
        return np.random.choice(wildcards)

# ================ CURRENT AFFAIRS ENGINE ================
class CurrentAffairsEngine:
    """Enhanced current affairs engine with Redis caching"""
    
    def __init__(self, db: ProductionDatabase):
        self.db = db
        self.current_events_entities = [
            "donald trump", "joe biden", "elon musk", "openai", "chatgpt",
            "taylor swift", "kanye west", "meta", "google", "tesla",
            "ukraine", "russia", "china", "ai", "climate change"
        ]
    
    async def fetch_context_for_comments(self, comments: str) -> Dict[str, str]:
        """Detect and fetch current context for entities in comments"""
        detected_entities = self._detect_entities(comments)
        
        if not detected_entities:
            return {}
        
        context = {}
        for entity in detected_entities:
            # Check Redis cache first
            cached = self.db.get_cached_affairs(entity)
            if cached:
                context[entity] = cached
            else:
                # Fetch fresh context and cache
                fresh_context = await self._search_current_context(entity)
                context[entity] = fresh_context
                self.db.cache_current_affairs(entity, fresh_context)
        
        return context
    
    def _detect_entities(self, comments: str) -> List[str]:
        """Detect current events entities in comments"""
        comments_lower = comments.lower()
        detected = []
        
        for entity in self.current_events_entities:
            if entity in comments_lower:
                detected.append(entity)
        
        return detected
    
    async def _search_current_context(self, entity: str) -> str:

# ================ DATABASE SETUP ================
class ComedyDatabase:
    def __init__(self, db_path="comedy_ai.db"):
        self.db_path = db_path
        self.init_database()
    
    def init_database(self):
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        # Comment batches with day.series format
        cursor.execute("""
        CREATE TABLE IF NOT EXISTS comment_batches (
            batch_id VARCHAR(10) PRIMARY KEY,
            comments TEXT NOT NULL,
            series_number INTEGER,
            day_number INTEGER,
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        )""")
        
        # Stage outputs for each batch
        cursor.execute("""
        CREATE TABLE IF NOT EXISTS stage_outputs (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            batch_id VARCHAR(10),
            stage_name VARCHAR(50),
            agent_name VARCHAR(50),
            input_text TEXT,
            output_text TEXT,
            feedback TEXT,
            optimal_response TEXT,
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        )""")
        
        # Self-improving prompts for each agent
        cursor.execute("""
        CREATE TABLE IF NOT EXISTS agent_prompts (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            agent_name VARCHAR(50),
            prompt_text TEXT,
            version_number INTEGER,
            improvement_notes TEXT,
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        )""")
        
        # Emotional state tracking
        cursor.execute("""
        CREATE TABLE IF NOT EXISTS emotional_states (
            batch_id VARCHAR(10),
            valence FLOAT,
            arousal FLOAT,
            dominant_emotion VARCHAR(50),
            stage VARCHAR(50),
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        )""")
        
        # User relationship tracking
        cursor.execute("""
        CREATE TABLE IF NOT EXISTS user_relationships (
            username VARCHAR(100),
            interaction_count INTEGER DEFAULT 0,
            relationship_type VARCHAR(50),
            last_roast TEXT,
            affection_level FLOAT,
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        )""")
        
        conn.commit()
        conn.close()

Here’s what that enables, based on the plan in ai_anthropologist_final_roadmap_v3:

State Management & Context Injection: The output of one agent is the input for the next, but it's not always a direct 1-to-1 pass.

The RelationalAgent needs the collection of all 5 scripts from the writers.

The TheoryAgent needs the single, cohesive draft produced by the RelationalAgent.

The PolishAgent needs the theoretically-framed script from the TheoryAgent.

BORIS is responsible for taking the output from one stage, packaging it correctly, and providing it as the precise input for the next stage.

Decision Making & Quality Control: What if the RelationalAgent produces a low-quality draft? In a simple pipe, the process would continue with flawed material. By returning to BORIS, the orchestrator can make decisions:

"The RelationalAgent's synthesis is weak. I will discard it and instead send the raw mocking script directly to the TheoryAgent because it was the strongest of the initial batch."

This allows for dynamic error handling and quality control at each step.

Phase-Aware Logic: As your roadmap outlines, the training is done in phases.

In Phase 1, the assembly line stops after the ComedyWriterAgents.

In Phase 2, it stops after the RelationalAgent.

Only a central orchestrator like BORIS can manage this conditional workflow. A simple, hard-coded pipe cannot adapt to the different training phases.

Modularity and Flexibility: This architecture makes the system incredibly easy to upgrade. Imagine you want to add a new FactCheckerAgent between the TheoryAgent and the PolishAgent.

With BORIS: You just modify BORIS's orchestration logic to add one more step to the assembly line. None of the other agents need to be changed.

Without BORIS (in a simple pipe): You would have to modify the TheoryAgent to call the FactCheckerAgent, and then modify the FactCheckerAgent to call the PolishAgent. This creates a brittle, tightly-coupled system that is difficult to maintain and extend.

In short, returning to BORIS at each step transforms the system from a dumb pipe into an intelligent, flexible, and robust creative workflow. BORIS is the "brain" that ensures the right work gets to the right specialist at the right time with the right context.
