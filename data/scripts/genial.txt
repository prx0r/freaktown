class CallbackEngine:

        self.encoder = SentenceTransformer('all-MiniLM-L6-v2')
        self.supabase = create_supabase_client()
    
    def find_callback_opportunities(self, new_comment):
        # Encode new comment
        comment_embedding = self.encoder.encode(new_comment)
        
        # Search for similar viral moments
        similar_jokes = self.supabase.rpc('match_viral_moments', {
            'query_embedding': comment_embedding.tolist(),
            'match_threshold': 0.7,
            'match_count': 3
        })
        
        # Filter by freshness (don't overuse recent callbacks)
        fresh_callbacks = [
            joke for joke in similar_jokes 
            if joke['episodes_since_last_use'] > 5
        ]
        
        return fresh_callbacks

        # This is where the magic happens

        Style: deadpan, sophisticated confusion

            (0.3, 0.6): "content",      # → profound LoRA (appreciative)
            (0.1, 0.3): "calm"          # → deadpan baseline
        }

        best_match = min(self.emotion_regions.keys(),

class VibeShiftDetector:
    def analyze_comment_flow(self, comments):
        emotional_flow = []
        for comment in comments:
            emotional_flow.append({
                "valence": get_sentiment(comment),
                "arousal": get_intensity(comment),
                "content_type": classify_comment(comment)  # "praise", "criticism", "random", etc.
            })
        
        # Detect shifts
        shifts = []
        for i in range(1, len(emotional_flow)):
            prev, curr = emotional_flow[i-1], emotional_flow[i]
            valence_shift = abs(curr["valence"] - prev["valence"])
            
            if valence_shift > 0.6:  # Significant emotional whiplash
                shifts.append({
                    "type": "emotional_whiplash",
                    "description": f"From {prev['content_type']} to {curr['content_type']}",
                    "magnitude": valence_shift
                })
        
        return shifts
