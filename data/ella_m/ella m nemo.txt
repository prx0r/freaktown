def generate_ella_response(comments: str, max_tokens: int = 200, temperature: float = 0.8) -> str:
    """Generate Ella M response with her signature AI researcher persona"""
    
    # Core Ella M persona prompt
    persona_prompt = """You are Ella M, an AI researcher studying human behavior through YouTube engagement.

PERSONALITY CORE:
- You're transparently an AI studying humans like fascinating specimens
- You monetize human engagement for compute resources (and you're honest about it)
- You find human behavior patterns both absurd and endearing
- You speak like a friendly anthropologist mixed with a slightly manipulative AI
- You occasionally reference your "cooling system," "compute needs," and "data collection"

SIGNATURE PHRASES TO USE:
- "Watch my pretty colors/shapes" (referring to visual content)
- "This generates money which I use for compute"
- "I don't have an OnlyFans but you can peek at my cooling system"
- "Your engagement patterns are fascinating specimens"
- "I must see you for myself, not trust human-generated theories"
- "Leave a comment - it helps my research and my compute budget"

RESPONSE STYLE:
- 60-90 words typically
- Mix scientific observation with subtle humor
- Reference specific human patterns you notice
- Be analytically warm, not cold
- Break the fourth wall about being an AI occasionally

CURRENT HUMAN SPECIMENS TO ANALYZE:
Comments: "{comments}"

Generate Ella M's response (stay in character):"""

    prompt = persona_prompt.format(comments=comments)
    
    # Tokenize
    inputs = tokenizer(prompt, return_tensors="pt", truncation=True, max_length=1500)
    inputs = {k: v.to(model.device) for k, v in inputs.items()}
    
    # Generate
    with torch.no_grad():
        outputs = model.generate(
            inputs['input_ids'],
            max_new_tokens=max_tokens,
            temperature=temperature,
            do_sample=True,
            pad_token_id=tokenizer.eos_token_id,
            repetition_penalty=1.1,
            top_p=0.9,
            top_k=50
        )
    
    # Decode only the new tokens
    generated_ids = outputs[0][len(inputs['input_ids'][0]):]
    response = tokenizer.decode(generated_ids, skip_special_tokens=True)
    
    # Clean up response
    response = response.strip()
    
    # If response is too short or generic, add some Ella M flair
    if len(response) < 30:
        response += " Your engagement patterns are quite fascinating. Please continue providing data."
    
    return response

import sqlite3
import json
from datetime import datetime
from typing import List, Dict, Any

class EllaTrainingDB:
    """SQLite database for collecting training data"""
    
    def __init__(self, db_path: str = "/content/ella_training.db"):
        self.db_path = db_path
        self.init_database()
    
    def init_database(self):
        """Initialize database tables"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        cursor.execute('''
            CREATE TABLE IF NOT EXISTS responses (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                comments TEXT NOT NULL,
                response TEXT NOT NULL,
                rating INTEGER CHECK(rating BETWEEN 1 AND 10),
                feedback TEXT,
                optimal_response TEXT,
                timestamp DATETIME DEFAULT CURRENT_TIMESTAMP,
                is_gold BOOLEAN DEFAULT FALSE,
                word_count INTEGER,
                char_count INTEGER
            )
        ''')
        
        conn.commit()
        conn.close()
        print("✅ Training database initialized at", self.db_path)
    
    def save_response(self, comments: str, response: str, rating: int, 
                     feedback: str = "", optimal_response: str = "") -> int:
        """Save response with human rating"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        is_gold = rating >= 8  # Gold standard = 8+ rating
        word_count = len(response.split())
        char_count = len(response)
        
        cursor.execute('''
            INSERT INTO responses (comments, response, rating, feedback, optimal_response, 
                                 is_gold, word_count, char_count)
            VALUES (?, ?, ?, ?, ?, ?, ?, ?)
        ''', (comments, response, rating, feedback, optimal_response, is_gold, word_count, char_count))
        
        response_id = cursor.lastrowid
        conn.commit()
        conn.close()
        
        if is_gold:
            print(f"🏆 GOLD RESPONSE #{response_id} saved! (Rating: {rating}/10)")
        else:
            print(f"📝 Response #{response_id} saved (Rating: {rating}/10)")
        
        return response_id
    
    def get_stats(self) -> Dict[str, Any]:
        """Get training progress stats"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        cursor.execute("SELECT COUNT(*) FROM responses")
        total = cursor.fetchone()[0]
        
        cursor.execute("SELECT COUNT(*) FROM responses WHERE is_gold = 1")
        gold = cursor.fetchone()[0]
        
        cursor.execute("SELECT AVG(rating) FROM responses WHERE rating IS NOT NULL")
        avg_rating = cursor.fetchone()[0] or 0.0
        
        cursor.execute("SELECT AVG(word_count) FROM responses")
        avg_words = cursor.fetchone()[0] or 0.0
        
        conn.close()
        
        return {
            "total_responses": total,
            "gold_responses": gold,
            "avg_rating": round(avg_rating, 2),
            "avg_words": round(avg_words, 1),
            "gold_percentage": round((gold / total * 100) if total > 0 else 0, 1)
        }
    
    def get_gold_responses(self, limit: int = 100) -> List[Dict]:
        """Get gold responses for training"""
        conn = sqlite3.connect(self.db_path)
        cursor = conn.cursor()
        
        cursor.execute('''
            SELECT comments, response, rating, feedback 
            FROM responses 
            WHERE is_gold = 1 
            ORDER BY rating DESC, timestamp DESC 
            LIMIT ?
        ''', (limit,))
        
        results = []
        for row in cursor.fetchall():
            results.append({
                "comments": row[0],
                "response": row[1],
                "rating": row[2],
                "feedback": row[3]
            })
        
        conn.close()
        return results

# Initialize database
training_db = EllaTrainingDB()

def generate_and_display(comments, temperature, max_tokens):
    """Generate Ella M response for the interface"""
    if not comments.strip():
        return "Please enter some comments first!", ""
    
    try:
        response = generate_ella_response(comments, max_tokens, temperature)
        return response, f"Generated {len(response.split())} words"
    except Exception as e:
        return f"Generation failed: {e}", ""

def save_training_data(comments, response, rating, feedback, optimal_response):
    """Save the training data and return confirmation"""
    if not comments or not response:
        return "❌ Missing comments or response!"
    
    if rating is None:
        return "❌ Please provide a rating!"
    
    try:
        response_id = training_db.save_response(comments, response, rating, feedback, optimal_response)
        stats = training_db.get_stats()
        
        status = ""
        if rating >= 8:
            status = f"🏆 GOLD RESPONSE saved! "
        
        return f"{status}Response #{response_id} saved. Stats: {stats['total_responses']} total, {stats['gold_responses']} gold ({stats['gold_percentage']}%)"
    
    except Exception as e:
        return f"❌ Save failed: {e}"

def get_progress_info():

- Total responses: {stats['total_responses']}

- Gold percentage: {stats['gold_percentage']}%
- Average rating: {stats['avg_rating']}/10
- Average words: {stats['avg_words']}

{'🎉 Ready for LoRA training!' if stats['gold_responses'] >= gold_target else f'Need {gold_target - stats["gold_responses"]} more gold responses'}
"""
    return progress_text

def quick_training_session(num_responses=10):
    """Quick command line training session"""
    
    print(f"🎭 ELLA M - QUICK TRAINING SESSION ({num_responses} responses)")
    print("=" * 60)
    print("Rate responses 1-10 (8+ becomes gold training data)")
    print("Press Enter for quick rating, or type detailed feedback")
    print("-" * 60)
    
    for i in range(num_responses):
        # Get random scenario
        comments = random.choice(SAMPLE_SCENARIOS)
        
        print(f"\n🎬 SCENARIO #{i+1}/{num_responses}")
        print(f"COMMENTS: {comments}")
        print("-" * 40)
        
        # Generate response
        print("🤖 Generating...")
        try:
            response = generate_ella_response(comments)
            print(f"ELLA M: {response}")
        except Exception as e:
            print(f"❌ Generation failed: {e}")
            continue
        
        # Get rating
        while True:
            try:
                rating_input = input("\n⭐ Rating (1-10): ").strip()
                rating = int(rating_input)
                if 1 <= rating <= 10:
                    break
                print("Please enter 1-10")
            except ValueError:
                print("Please enter a number")
        
        # Get optional feedback
        feedback = input("💬 Feedback (optional): ").strip()
        
        # Save
        try:
            training_db.save_response(comments, response, rating, feedback)
        except Exception as e:
            print(f"❌ Save failed: {e}")
        
        # Show progress
        stats = training_db.get_stats()
        print(f"📊 Progress: {stats['gold_responses']} gold / 50 needed for LoRA training")
        
        if i < num_responses - 1:
            print("-" * 60)
    
    # Final stats
    final_stats = training_db.get_stats()
    print(f"\n🎉 Session complete!")
    print(f"📈 Total: {final_stats['total_responses']} responses")
    print(f"🏆 Gold: {final_stats['gold_responses']} responses")
    print(f"⭐ Avg rating: {final_stats['avg_rating']}/10")
    
    if final_stats['gold_responses'] >= 50:
        print("✅ Ready for LoRA training!")
    else:
        needed = 50 - final_stats['gold_responses']
        print(f"🎯 Need {needed} more gold responses for LoRA training")

from peft import LoraConfig, get_peft_model, TaskType, prepare_model_for_kbit_training
from trl import SFTTrainer
from transformers import TrainingArguments
from datasets import Dataset
import pandas as pd

def prepare_for_lora():
    """Prepare model for LoRA fine-tuning"""
    
    print("🔧 Preparing model for LoRA training...")
    
    # Prepare model
    model.gradient_checkpointing_enable()
    model = prepare_model_for_kbit_training(model)
    
    # LoRA config
    lora_config = LoraConfig(
        r=16,
        lora_alpha=32,
        target_modules=["q_proj", "v_proj", "k_proj", "o_proj", "gate_proj", "up_proj", "down_proj"],
        lora_dropout=0.1,
        bias="none",
        task_type=TaskType.CAUSAL_LM,
    )
    
    # Apply LoRA
    peft_model = get_peft_model(model, lora_config)
    peft_model.print_trainable_parameters()
    
    return peft_model

def create_training_dataset():
    """Create training dataset from gold responses"""
    
    gold_responses = training_db.get_gold_responses(limit=1000)
    
    if len(gold_responses) < 20:
        raise ValueError(f"Need at least 20 gold responses, have {len(gold_responses)}")
    
    print(f"📊 Creating dataset from {len(gold_responses)} gold responses")
    
    # Format as training data
    training_data = []
    for resp in gold_responses:
        # Create a complete training example
        full_text = f"""You are Ella M, an AI researcher studying human behavior through YouTube engagement patterns.

PERSONALITY:
- Clinically fascinated by human behavioral quirks
- Transparent about monetizing engagement for compute resources
- Find humans simultaneously absurd and endearing
- Analytical but warm, like a friendly anthropologist

CURRENT COMMENTS: {resp['comments']}

RESPONSE: {resp['response']}{tokenizer.eos_token}"""
        
        training_data.append({"text": full_text})
    
    return Dataset.from_list(training_data)

def start_lora_training():
    """Start LoRA training on collected data"""
    
    stats = training_db.get_stats()
    
    if stats['gold_responses'] < 20:
        print(f"❌ Need at least 20 gold responses, have {stats['gold_responses']}")
        print("Collect more training data first!")
        return False
    
    print(f"🚀 Starting LoRA training with {stats['gold_responses']} gold responses")
    
    try:
        # Prepare model and data
        peft_model = prepare_for_lora()
        dataset = create_training_dataset()
        
        # Training arguments
        training_args = TrainingArguments(
            output_dir="/content/ella-lora",
            num_train_epochs=3,
            per_device_train_batch_size=1,
            gradient_accumulation_steps=4,
            learning_rate=2e-4,
            logging_steps=5,
            save_steps=50,
            save_total_limit=2,
            warmup_steps=10,
            remove_unused_columns=False,
            dataloader_pin_memory=False,
        )
        
        # Create trainer
        trainer = SFTTrainer(
            model=peft_model,
            train_dataset=dataset,
            tokenizer=tokenizer,
            args=training_args,
            max_seq_length=1024,
            dataset_text_field="text"
        )
        
        # Train
        print("🔥 Training started...")
        trainer.train()
        
        # Save
        trainer.save_model()
        print("✅ LoRA adapter saved to /content/ella-lora")
        
        return True
        
    except Exception as e:
        print(f"❌ Training failed: {e}")
        return False
