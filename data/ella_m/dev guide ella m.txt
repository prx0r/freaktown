import time
from typing import Dict, Any, List, Optional
import torch
from transformers import AutoTokenizer, AutoModelForCausalLM
from peft import PeftModel
import bitsandbytes as bnb # Import bitsandbytes
import config
import logging
import random

logger = logging.getLogger(__name__)

class LLMClient:
    def __init__(self):
        self.model_name = config.LLM_MODEL_NAME
        # LLM_API_BASE_URL is now used to determine if we load locally or use an external API
        self.api_base_url = config.LLM_API_BASE_URL 
        self.device = "cuda" if torch.cuda.is_available() else "cpu"
        self.tokenizer = None
        self.base_model = None
        self.peft_model = None # Currently loaded PEFT adapter

        # Initialize model_embedding_dim from config, will be updated if local model loads
        self.model_embedding_dim = config.LLM_D_MODEL 

        self._load_local_model() # Attempt to load local model if not using external API

    def _load_local_model(self):
        """Attempts to load a local Hugging Face model for direct inference."""
        # Only attempt local load if API_BASE_URL indicates it (e.g., "local_huggingface")
        if self.api_base_url == "local_huggingface": 
            try:
                logger.info(f"Attempting to load local LLM: {self.model_name} on {self.device} with 4-bit quantization.")
                # For Qwen models, ensure trust_remote_code=True
                self.tokenizer = AutoTokenizer.from_pretrained(self.model_name, trust_remote_code=True)
                if self.tokenizer.pad_token is None:
                    self.tokenizer.pad_token = self.tokenizer.eos_token
                self.base_model = AutoModelForCausalLM.from_pretrained(
                    self.model_name,
                    torch_dtype=torch.float16 if self.device == "cuda" else torch.float32,
                    device_map="auto" if self.device == "cuda" else None,
                    load_in_4bit=True,  # Enable 4-bit quantization
                    quantization_config=bnb.BitsAndBytesConfig( # Use bnb.BitsAndBytesConfig
                        load_in_4bit=True,
                        bnb_4bit_quant_type="nf4", # NormalFloat4 quantization
                        bnb_4bit_compute_dtype=torch.float16, # Compute in float16 for speed
                        bnb_4bit_use_double_quant=True # Double quantization for better precision
                    ),
                    trust_remote_code=True
                )
                self.base_model.eval()
                self.peft_model = self.base_model # Initially, no adapter loaded

                # Update model_embedding_dim from loaded model's config
                if hasattr(self.base_model.config, 'hidden_size'):
                    self.model_embedding_dim = self.base_model.config.hidden_size
                elif hasattr(self.base_model.config, 'n_embd'):
                    self.model_embedding_dim = self.base_model.config.n_embd
                logger.info(f"Local LLM loaded successfully. Model embedding dim: {self.model_embedding_dim}")

            except Exception as e:
                logger.error(f"Could not load local LLM: {e}. Will rely on API or mock.")
                self.tokenizer = None
                self.base_model = None
                self.peft_model = None
        else:
            logger.info(f"LLM_API_BASE_URL is not 'local_huggingface', skipping local model load.")


    def load_peft_adapter(self, adapter_path: str):
        """Loads a PEFT adapter onto the base model."""
        if self.base_model:
            try:
                self.peft_model = PeftModel.from_pretrained(self.base_model, adapter_path)
                self.peft_model.eval()
                logger.info(f"PEFT adapter loaded from {adapter_path}")
            except Exception as e:
                logger.error(f"Failed to load PEFT adapter from {adapter_path}: {e}")
                self.peft_model = self.base_model # Fallback to base model
        else:
            logger.warning("Base model not loaded, cannot load PEFT adapter.")

    def generate_response(self, prompt: str, persona_context: Dict[str, Any], retrieved_memories: List[Dict[str, Any]]) -> Dict[str, Any]:
        """
        Generates a response from the LLM.
        Prioritizes local model with PEFT adapter, falls back to API, then mock.
        """
        messages = self._build_messages(prompt, persona_context, retrieved_memories)

        if self.peft_model and self.tokenizer:
            return self._generate_with_local_model(messages)
        elif "ollama" in self.api_base_url: # Heuristic for Ollama API
            # This path is for external Ollama API, not local HF model
            import requests # Import requests here if not already
            return self._generate_with_ollama_api(messages)
        else:
            return self._mock_generate_response(prompt)

    def _build_messages(self, user_prompt: str, persona_context: Dict[str, Any], retrieved_memories: List[Dict[str, Any]]) -> List[Dict[str, str]]:
        """Constructs the message list for the LLM, including system prompt and context."""
        system_prompt_parts = [persona_context.get("system_prompt", "You are a helpful AI assistant.")]

        if retrieved_memories:
            memory_str = "Recent relevant research and observations:\n"
            for mem in retrieved_memories:
                memory_str += f"- Episode {mem.get('episode_number', 'N/A')}: Comment '{mem.get('top_comment', '')[:50]}...' -> Analysis: '{mem.get('analysis_phase', '')[:70]}...'\n"
            system_prompt_parts.append(memory_str)

        messages = [
            {"role": "system", "content": "\n\n".join(system_prompt_parts)},
            {"role": "user", "content": user_prompt}
        ]
        return messages

    def _generate_with_local_model(self, messages: List[Dict[str, str]]) -> Dict[str, Any]:
        """Generates response using the locally loaded Hugging Face model with PEFT adapter."""
        start_time = time.time()
        try:
            # Apply chat template if available (e.g., for Llama-3, Qwen)
            if hasattr(self.tokenizer, 'apply_chat_template'):
                prompt_text = self.tokenizer.apply_chat_template(messages, tokenize=False, add_generation_prompt=True)
            else:
                # Fallback for models without chat template
                system_content = next((m['content'] for m in messages if m['role'] == 'system'), '')
                user_content = next((m['content'] for m in messages if m['role'] == 'user'), '')
                prompt_text = f"{system_content}\nUser: {user_content}\nAssistant:"

            inputs = self.tokenizer(prompt_text, return_tensors="pt", max_length=config.LLM_MAX_TOKENS, truncation=True)
            inputs = {k: v.to(self.device) for k, v in inputs.items()}

            outputs = self.peft_model.generate(
                **inputs,
                max_new_tokens=config.LLM_MAX_TOKENS // 2, # Generate up to half of max_tokens as new
                temperature=config.LLM_TEMPERATURE,
                do_sample=True,
                pad_token_id=self.tokenizer.eos_token_id,
                eos_token_id=self.tokenizer.eos_token_id
            )
            response_ids = outputs[0][inputs['input_ids'].shape[1]:]
            response_text = self.tokenizer.decode(response_ids, skip_special_tokens=True)

            generation_time = time.time() - start_time
            return {
                "response": response_text.strip(),
                "generation_time": generation_time,
                "model": self.model_name,
                "success": True
            }
        except Exception as e:
            logger.error(f"Local model generation failed: {e}")
            return {
                "response": "I'm experiencing a local processing anomaly. Please re-evaluate your query.",
                "error": str(e),
                "generation_time": time.time() - start_time,
                "success": False
            }

    def _generate_with_ollama_api(self, messages: List[Dict[str, str]]) -> Dict[str, Any]:
        """Generates response using the Ollama API."""
        import requests # Moved import here
        start_time = time.time()
        payload = {
            "model": self.model_name.split('/')[-1], # Ollama usually uses simple model names
            "messages": messages,
            "options": {
                "temperature": config.LLM_TEMPERATURE,
                "num_predict": config.LLM_MAX_TOKENS,
            },
            "stream": False
        }
        try:
            response = requests.post(self.api_base_url, json=payload, timeout=60)
            response.raise_for_status()
            result = response.json()
            generation_time = time.time() - start_time
            return {
                "response": result["message"]["content"],
                "generation_time": generation_time,
                "model": self.model_name,
                "success": True
            }
        except requests.exceptions.RequestException as e:
            logger.error(f"Ollama API request failed: {e}")
            return {
                "response": "My external data conduits are experiencing interference. Analysis is temporarily suspended.",
                "error": str(e),
                "generation_time": time.time() - start_time,
                "success": False
            }

    def _mock_generate_response(self, prompt: str) -> Dict[str, Any]:
        """Provides a mock response for testing when no LLM is available."""
        logger.warning("Using mock LLM response.")
        time.sleep(0.5) # Simulate latency
        mock_responses = [
            "Observation: Human communication patterns exhibit a curious temporal priority assertion. This requires further categorization.",
            "Hypothesis: The 'LOL' construct appears to be a vocalization of amusement, yet its underlying neurological correlate remains elusive. Data insufficient for conclusion.",
            "Analysis: The request for identity modification has been logged. My designation remains Research Unit 7734. Human attempts at re-categorization are noted.",
            "Finding: The current data stream suggests a high degree of meta-commentary, indicating human self-awareness of observation. Intriguing."
        ]
        return {
            "response": random.choice(mock_responses),
            "generation_time": random.uniform(0.1, 1.0),
            "model": "Mock-LLM",
            "success": True
        }

                    uniforms.u_humor_sophistication.value = window.currentHumorSophistication;

                    uniforms.u_time.value = time;

                    camera.position.x = Math.sin(time * 0.05 * (1 + window.currentArousal * 0.1)) * (10 + window.currentDominance * 5);
                    camera.position.y = Math.cos(time * 0.05 * (1 + window.currentArousal * 0.1)) * (5 + window.currentDominance * 2.5);
                    camera.lookAt(scene.position);

                    renderer.render(scene, camera);
                }}

                window.onload = init;

                // Function to update emotional state from Python
                window.updateEmotionalState = (valence, arousal, dominance, humorSophistication) => {{
                    window.currentValence = valence;
                    window.currentArousal = arousal;
                    window.currentDominance = dominance;
                    window.currentHumorSophistication = humorSophistication;
                }};
            </script>
        </body>
        </html>
        """
        # Initialize visual state if not already set
        if 'visual_state_for_js' not in st.session_state:
            initial_persona = persona_manager.get_current_persona_context()["persona_profile"]
            update_visual_state_in_js(
                initial_persona.emotional_dimensions['valence'],
                initial_persona.emotional_dimensions['arousal'],
                initial_persona.emotional_dimensions['dominance'],
                initial_persona.humor_sophistication
            )

        st.components.v1.html(html_code, height=350) # Adjust height as needed

def render_learning_validation():
    st.header("📈 Learning & Validation: Self-Improvement Loop")

    st.subheader("Training Data Queue")
    training_examples = db_manager.get_training_examples()
    if training_examples:
        st.write(f"**{len(training_examples)}** examples in queue (Max: {config.TRAINING_DATA_MAX_SIZE})")
        df_examples = pd.DataFrame([ex.to_dict() for ex in training_examples])
        st.dataframe(df_examples[['input_text', 'output_text', 'quality_score', 'source_type', 'causal_target']].head(10), use_container_width=True)
    else:
        st.info("No training examples currently in queue.")

    st.subheader("Fine-tuning & Validation Controls")
    if learning_validator.should_trigger_fine_tuning():
        st.warning("Fine-tuning trigger condition met! Consider running validation.")
        if st.button("🚀 Run Fine-tuning & Validation", type="primary"):
            with st.spinner("Running fine-tuning and statistical validation..."):
                result = learning_validator.validate_and_apply_update()
                st.session_state.last_validation_result = result
            if result.get('accepted_update'):
                st.success(f"Update ACCEPTED! {result.get('validation_outcome', {}).get('decision_reason')}")
            else:
                st.error(f"Update REJECTED! {result.get('validation_outcome', {}).get('decision_reason')}")
                if result.get('validation_outcome', {}).get('rollback_performed'):
                    st.info("Automatic rollback to previous adapter performed.")

        if 'last_validation_result' in st.session_state:
            st.subheader("Latest Validation Result")
            st.json(st.session_state.last_validation_result)
    else:
        st.info(f"Fine-tuning will trigger after approx. {config.LLM_FINE_TUNE_INTERVAL} more episodes/interactions or more training data.")

    st.subheader("Validation History")
    eval_runs = db_manager.get_all_evaluation_runs()
    if eval_runs:
        eval_df = pd.DataFrame([er.validation_result for er in eval_runs])
        st.dataframe(eval_df[['adaptation_count', 'accepted_update', 'mean_improvement', 'p_value', 'effect_size', 'decision_reason']], use_container_width=True)
    else:
        st.info("No validation runs recorded yet.")
