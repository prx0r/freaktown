-- Core persona definitions tailored for Ella M.
CREATE TABLE personas (
    id SERIAL PRIMARY KEY,
    persona_name VARCHAR(100) UNIQUE NOT NULL,
    profile_text TEXT NOT NULL,
    creation_date TIMESTAMP DEFAULT NOW(),
    is_dynamic BOOLEAN DEFAULT FALSE,
    is_active BOOLEAN DEFAULT TRUE,
    enhancement_suggestions TEXT[] DEFAULT '{}',
    last_enhanced TIMESTAMP
);

-- Topics table for extensible categories
CREATE TABLE topics (
    id SERIAL PRIMARY KEY,
    topic_name VARCHAR(100) UNIQUE NOT NULL,
    description TEXT,
    created_at TIMESTAMP DEFAULT NOW()
);

-- Users table for tracking comment history
CREATE TABLE users (
    id SERIAL PRIMARY KEY,
    username VARCHAR(100) UNIQUE NOT NULL,
    first_comment_date TIMESTAMP,
    total_comments INT DEFAULT 0,
    engagement_score FLOAT DEFAULT 0.0,
    context_tags TEXT[] DEFAULT '{}',
    last_active TIMESTAMP DEFAULT NOW()
);

-- Comments table for original user inputs with extensible topics
CREATE TABLE comments (
    id SERIAL PRIMARY KEY,
    user_id INT REFERENCES users(id),
    comment_text TEXT NOT NULL,
    created_at TIMESTAMP DEFAULT NOW(),
    topic_id INT REFERENCES topics(id),
    engagement_score FLOAT DEFAULT 0.0
);

-- Scout's instructions to Ella
CREATE TABLE scout_instructions (
    id SERIAL PRIMARY KEY,
    persona_id INT REFERENCES personas(id),
    instruction_text TEXT NOT NULL,
    created_at TIMESTAMP DEFAULT NOW(),
    target_comment_id INT REFERENCES comments(id),
    enhancement_applied TEXT
);

-- All responses (exemplars + generated + posted) for Ella M.
CREATE TABLE responses (
    id SERIAL PRIMARY KEY,
    persona_id INT REFERENCES personas(id),
    comment_id INT REFERENCES comments(id),
    instruction_id INT REFERENCES scout_instructions(id),
    response_text TEXT NOT NULL,
    engagement_score FLOAT DEFAULT 0.0,
    youtube_views INT DEFAULT 0,
    youtube_likes INT DEFAULT 0,
    youtube_comments INT DEFAULT 0,
    engagement_rate FLOAT DEFAULT 0.0,
    used_in_video BOOLEAN DEFAULT FALSE,
    times_referenced INT DEFAULT 0,
    is_exemplar BOOLEAN DEFAULT FALSE,
    video_post_date TIMESTAMP,
    enhancement_applied TEXT,
    user_context TEXT,
    optimal_timing VARCHAR(50),
    response_length INT,
    content_hash VARCHAR(64) UNIQUE,
    callback_references TEXT[] DEFAULT '{}',
    created_at TIMESTAMP DEFAULT NOW()
);

-- Gold table for viral/high-engagement videos
CREATE TABLE gold_videos (
    id SERIAL PRIMARY KEY,
    response_id INT REFERENCES responses(id),
    instruction_id INT REFERENCES scout_instructions(id),
    persona_name VARCHAR(100) NOT NULL,
    original_comment_id INT REFERENCES comments(id),
    user_id INT REFERENCES users(id),
    response_text TEXT NOT NULL,
    engagement_score FLOAT NOT NULL CHECK (engagement_score >= 8.0),
    youtube_views INT NOT NULL,
    youtube_likes INT NOT NULL,
    youtube_comments INT NOT NULL,
    engagement_rate FLOAT NOT NULL,
    video_post_date TIMESTAMP,
    analysis_notes TEXT,
    created_at TIMESTAMP DEFAULT NOW()
);

-- Scout's engagement research and learning for Ella M.
CREATE TABLE engagement_research (
    id SERIAL PRIMARY KEY,
    research_category VARCHAR(100),
    insight_title VARCHAR(200) NOT NULL,
    insight_description TEXT NOT NULL,
    supporting_data JSONB,
    confidence_score FLOAT CHECK (confidence_score >= 0 AND confidence_score <= 1),
    sample_size INT DEFAULT 0,
    user_quotes TEXT[] DEFAULT '{}',
    correlation_strength FLOAT,
    discovery_date TIMESTAMP DEFAULT NOW(),
    last_validated TIMESTAMP,
    research_status VARCHAR(50) DEFAULT 'hypothesis' CHECK (research_status IN ('hypothesis', 'confirmed', 'disproven')),
    times_referenced INT DEFAULT 0,
    tags TEXT[] DEFAULT '{}'
);

-- Performance indexes
CREATE INDEX idx_responses_engagement ON responses(engagement_score DESC, persona_id);
CREATE INDEX idx_responses_created_at ON responses(created_at DESC);
CREATE INDEX idx_responses_persona_active ON responses(persona_id) WHERE used_in_video = FALSE;
CREATE INDEX idx_users_active ON users(last_active DESC);
CREATE INDEX idx_comments_user ON comments(user_id, created_at DESC);
CREATE INDEX idx_scout_instructions_target ON scout_instructions(target_comment_id);
CREATE INDEX idx_gold_videos_engagement ON gold_videos(engagement_score DESC);
CREATE INDEX idx_research_category ON engagement_research(research_category, confidence_score DESC);
