class RedisClient:
    def __init__(self, host='localhost', port=6379, db=0):
        self.client = redis.StrictRedis(host=host, port=port, db=db, decode_responses=True)
        self.user_id = "single_user_id" # Hardcoded for single-user system

    def _key(self, suffix: str) -> str:
        return f"{self.user_id}:{suffix}"

    def set_relational_context_vector(self, vector: list[float]):
        self.client.set(self._key("relational_context_vector"), json.dumps(vector))

    def get_relational_context_vector(self) -> list[float] | None:
        data = self.client.get(self._key("relational_context_vector"))
        return json.loads(data) if data else None

    # Now specifically for Momentary Modulation LoRA
    def set_momentary_lora_state(self, lora_weights_b64: str): # Base64 encoded LoRA weights
        self.client.set(self._key("momentary_lora_state"), lora_weights_b64)

    def get_momentary_lora_state(self) -> str | None:
        return self.client.get(self._key("momentary_lora_state"))

    def set_user_model_snapshot(self, user_model: Dict[str, Any]):
        self.client.set(self._key("user_model_snapshot"), json.dumps(user_model))

    def get_user_model_snapshot(self) -> Dict[str, Any] | None:
        data = self.client.get(self._key("user_model_snapshot"))
        return json.loads(data) if data else None

    def clear_state(self):
        # For development/testing, clear all state for the user
        self.client.delete(self._key("relational_context_vector"))
        self.client.delete(self._key("momentary_lora_state"))
        self.client.delete(self._key("user_model_snapshot"))

@app.post("/interact", response_model=InteractResponse)
async def interact(request: InteractRequest):
    try:
        # 1. Input Reception & Preprocessing
        audio_bytes = base64.b64decode(request.audio_input_b64)
        behavioral_signature = await audio_extractor.process_audio(audio_bytes)
        user_input_text = behavioral_signature.text
        user_input_prosody_vector = behavioral_signature.prosody_vector

        # 2. Pillar 1: The Analyst Core (The "Observer")
        user_model = redis_client.get_user_model_snapshot() or analyst_core.get_default_user_model()
        updated_user_model = await analyst_core.update_user_model(behavioral_signature, user_model)
        redis_client.set_user_model_snapshot(updated_user_model)

        context_package = await analyst_core.perform_dual_retrieval(
            user_input_text, updated_user_model
        )

        # 3. Pillar 2: The LinOSS Relational Core (The "Atmosphere")
        relational_context_vector = redis_client.get_relational_context_vector()
        updated_relational_context_vector = await linoss_model.update_state(
            behavioral_signature, relational_context_vector
        )
        redis_client.set_relational_context_vector(updated_relational_context_vector)

        # 4. Pillar 3: The Agentic Core (The "Actor")
        # Get current Momentary LoRA state (weights)
        current_momentary_lora_weights_b64 = redis_client.get_momentary_lora_state()
        current_momentary_lora_weights = None
        if current_momentary_lora_weights_b64:
            current_momentary_lora_weights = LoRAWeights.from_bytes(base64.b64decode(current_momentary_lora_weights_b64))

        draft_response_text, new_momentary_lora_weights = await agentic_core.generate_response(
            updated_user_model,
            context_package,
            updated_relational_context_vector,
            current_momentary_lora_weights
        )

        # Persist new Momentary LoRA state hot
        new_momentary_lora_weights_b64 = base64.b64encode(new_momentary_lora_weights.to_bytes()).decode('utf-8')
        redis_client.set_momentary_lora_state(new_momentary_lora_weights_b64)

        # 5. Output Generation & Post-processing
        audio_response_bytes = await TTSSystem.synthesize_speech(draft_response_text)
        audio_response_b64 = base64.b64encode(audio_response_bytes).decode('utf-8')

        # 6. Asynchronous Learning & Persistence (Fire and Forget)
        asyncio.create_task(
            db_manager.log_interaction(
                user_input_text, user_input_prosody_vector,
                draft_response_text, audio_extractor.extract_prosody_from_text(draft_response_text),
                updated_relational_context_vector, updated_user_model,
                "inferred_intent_placeholder",
                audio_extractor.get_semantic_embedding(user_input_text + " " + draft_response_text)
            )
        )

        return InteractResponse(
            audio_response_b64=audio_response_b64,
            debug_info={
                "user_model_snapshot": updated_user_model,
                "relational_context_vector_snapshot": updated_relational_context_vector.tolist(),
                "context_package_summary": {k: str(v)[:50] for k, v in context_package.items()} # Truncate for debug
            }
        )

    except Exception as e:
        print(f"Error during interaction: {e}")
        raise HTTPException(status_code=500, detail=f"Internal server error: {str(e)}")

class LinOSSModel(nn.Module):
    def __init__(self, input_dim: int = 1536 + 256, state_dim: int = 512):
        super().__init__()
        self.state_dim = state_dim
        self.embedding_model = EmbeddingModel() # Initialize EmbeddingModel

        # Damped LinOSS parameters (learned through training)
        # A: State transition matrix (controls decay and interaction between state components)
        # B: Input matrix (how input affects state)
        self.A = nn.Parameter(torch.randn(state_dim, state_dim))
        self.B = nn.Parameter(torch.randn(input_dim, state_dim))
        # For Damped LinOSS, 'A' would be structured to incorporate learned decay rates
        # (e.g., via a diagonal matrix for damping and a skew-symmetric matrix for oscillation).
        # This is a simplified representation.

        # Initial state (can be learned or zero-initialized)
        self._current_state = nn.Parameter(torch.zeros(state_dim), requires_grad=False)

    async def load_initial_state(self):
        # Load last known state from Redis or DB
        saved_state = redis_client.get_relational_context_vector()
        if saved_state:
            self._current_state.data = torch.tensor(saved_state).float()
            print(f"LinOSS state loaded from Redis: {self._current_state.mean().item():.4f}")
        else:
            print("LinOSS state initialized to zero.")

    async def persist_state_to_db(self):
        # This would be part of AsyncStatePersistor, called to save to PostgreSQL
        # For shutdown, ensure it's saved to Redis first, then AsyncStatePersistor handles DB.
        redis_client.set_relational_context_vector(self._current_state.tolist())

    async def update_state(self, signature: 'BehavioralSignature', current_state: Optional[List[float]] = None) -> List[float]:
        # Convert signature text to embedding
        text_embedding = await self.embedding_model.get_embedding(signature.text)
        
        # Combine text embedding and prosody vector
        input_features = torch.cat([torch.tensor(text_embedding), torch.tensor(signature.prosody_vector)]).float()

        # Load current state if provided, otherwise use internal
        if current_state is not None:
            self._current_state.data = torch.tensor(current_state).float()

        # LinOSS state update equation (simplified)
        # h_t = A * h_{t-1} + B * x_t
        # For Damped LinOSS, the A matrix would incorporate learned decay rates.
        new_state = torch.matmul(self._current_state, self.A) + torch.matmul(input_features, self.B)
        self._current_state.data = new_state # Update internal state for next turn

        redis_client.set_relational_context_vector(new_state.tolist()) # Persist hot to Redis
        return new_state.tolist()

    def get_state(self) -> List[float]:
        return self._current_state.tolist()

# LoRAWeights class remains the same for both Base and Momentary LoRAs
class LoRAWeights:
    def __init__(self, W_A: torch.Tensor, W_B: torch.Tensor):
        self.W_A = W_A # LoRA A matrix
        self.W_B = W_B # LoRA B matrix

    def to_bytes(self) -> bytes:
        buffer = io.BytesIO()
        torch.save({'W_A': self.W_A, 'W_B': self.W_B}, buffer)
        return buffer.getvalue()

    @staticmethod
    def from_bytes(data: bytes) -> 'LoRAWeights':
        buffer = io.BytesIO(data)
        loaded = torch.load(buffer)
        return LoRAWeights(loaded['W_A'], loaded['W_B'])

class HyperNetwork(nn.Module):
    def __init__(self, input_dim: int, output_dim_lora_a: int, output_dim_lora_b: int):
        super().__init__()
        # Input: Concatenation of UserModel, Context Package embeddings, RelationalContextVector
        # Output: Delta weights for MOMENTARY LoRA A and B matrices
        
        self.fc1 = nn.Linear(input_dim, 1024)
        self.relu = nn.ReLU()
        self.fc_lora_a = nn.Linear(1024, output_dim_lora_a)
        self.fc_lora_b = nn.Linear(1024, output_dim_lora_b)
        
        # Apply tanh to bound outputs for stability
        self.tanh_a = nn.Tanh()
        self.tanh_b = nn.Tanh()
        
        # Strong regularization (e.g., dropout) would be applied during training
        self.dropout = nn.Dropout(0.1)

    def forward(self, x: torch.Tensor) -> Dict[str, torch.Tensor]:
        x = self.dropout(self.relu(self.fc1(x)))
        delta_W_A = self.tanh_a(self.fc_lora_a(x))
        delta_W_B = self.tanh_b(self.fc_lora_b(x))
        return {'delta_W_A': delta_W_A, 'delta_W_B': delta_W_B}

            self._current_momentary_lora_weights = LoRAWeights.from_bytes(base64.b64decode(lora_weights_b64))
            print("Momentary LoRA state loaded from Redis.")
        else:
            # Initialize with zero LoRA weights if no state found
            self._current_momentary_lora_weights = LoRAWeights(
                torch.zeros(self.llm_hidden_dim, self.lora_rank),
                torch.zeros(self.lora_rank, self.llm_hidden_dim)
            )
            print("Momentary LoRA state initialized to zero.")

    async def persist_state_to_db(self):
        # This would be part of AsyncStatePersistor
        if self._current_momentary_lora_weights:
            redis_client.set_momentary_lora_state(
                base64.b64encode(self._current_momentary_lora_weights.to_bytes()).decode('utf-8')
            )

    def _apply_recurrence(self, delta_W_A: torch.Tensor, delta_W_B: torch.Tensor) -> LoRAWeights:
        # Reshape delta_W_A and delta_W_B to their matrix forms
        delta_W_A_reshaped = delta_W_A.view(self.llm_hidden_dim, self.lora_rank)
        delta_W_B_reshaped = delta_W_B.view(self.lora_rank, self.llm_hidden_dim)

        if self._current_momentary_lora_weights:
            # ΔW_Mod_t = (decay * A) @ ΔW_Mod_{t-1} + B @ δ_t
            # This recurrence needs careful design. Here's a simplified conceptual one:
            new_W_A = self.decay_rate * self._current_momentary_lora_weights.W_A + delta_W_A_reshaped
            new_W_B = self.decay_rate * self._current_momentary_lora_weights.W_B + delta_W_B_reshaped
        else:
            new_W_A = delta_W_A_reshaped
            new_W_B = delta_W_B_reshaped
            
        return LoRAWeights(new_W_A, new_W_B)

    async def generate_and_update_lora(self, combined_input_features: torch.Tensor) -> LoRAWeights:
        delta_weights = self.hypernetwork(combined_input_features)
        new_lora_weights = self._apply_recurrence(
            delta_weights['delta_W_A'], delta_weights['delta_W_B']
        )
        self._current_momentary_lora_weights = new_lora_weights
        return new_lora_weights

    async def check_coherence(self, draft_response: str, recent_ai_statements: List[str], user_beliefs: List[Dict[str, Any]]) -> bool: # user_beliefs now Dict[str, Any]
        response_embedding = await self.embedding_model.get_embedding(draft_response)

        # Check against AI's own recent statements
        for statement in recent_ai_statements:
            statement_embedding = await self.embedding_model.get_embedding(statement)
            similarity = torch.cosine_similarity(torch.tensor(response_embedding), torch.tensor(statement_embedding), dim=0)
            # A low similarity or specific NLI contradiction score would indicate incoherence
            if similarity < 0.2: # Example threshold for potential contradiction
                print(f"CoherenceGuardian: Potential self-contradiction detected with statement: '{statement}'")
                return False

        # Check against user's core beliefs
        for belief in user_beliefs:
            belief_embedding = torch.tensor(belief['embedding']) # Use pre-computed embedding from DB
            similarity = torch.cosine_similarity(torch.tensor(response_embedding), belief_embedding, dim=0)
            # A high negative similarity or NLI contradiction score would indicate violation
            if similarity < 0.1: # Example threshold for potential belief violation
                print(f"CoherenceGuardian: Potential user belief violation detected with belief: '{belief['belief_text']}'")
                return False

        return True # All checks passed

Phase 1: After a Day of Interaction (Short-Term Dynamic Adaptation & Initial Learning)
You've just finished your first day of interacting with Samantha.

What's Happening in Real-Time (Per Turn):

Your Voice Enters the System:

Your spoken audio stream hits the FastAPI /interact endpoint.

The AudioBasedSignatureExtractor (WhisperX for ASR, a dedicated module for prosody features) converts your audio into a BehavioralSignature (text transcript + prosody_vector). This is milliseconds-level processing.

Pillar 1: The Analyst Core - Understanding You (Initial Insights):

The AnalystCore receives your BehavioralSignature.

Its AnalystModel (Mamba/LinOSS based, pre-trained on IPIP synthetic data with GAN refinement) processes your BehavioralSignature. It immediately starts inferring your IPIP personality facet scores, subtly adjusting your UserModel (e.g., if your voice is high-energy, your "cheerfulness" score might increase for the session). This UserModel is hot-stored in Redis.

The BeliefExtractor (an LLM-driven process) might extract initial explicit beliefs or values from your conversational text. These are added to your BeliefGraph in PostgreSQL.

The Dual-Retrieval Engine starts querying your (still small) Episodic Memory (the interaction_log in PostgreSQL) for similar past interactions, and your (nascent) BeliefGraph for relevant core beliefs. At Day 1, these retrievals are basic, but they begin building foundational context.

Pillar 2: The LinOSS Relational Core - Sensing the "Vibe":

The DampedLinOSSModel receives your BehavioralSignature and its current RelationalContextVector (RCV) from Redis.

It performs a state update (h_t = A*h_{t-1} + B*x_t), integrating your input into the RCV. The RCV immediately reflects the "vibe" of your current turn (e.g., if you're playful, the RCV shifts to a more "light" state).

The updated RelationalContextVector is immediately hot-stored back into Redis.

Pillar 3: The Agentic Core - Adapting & Responding (Dual-LoRA in Action):

The HyperNetwork receives a combined input of your UserModel, the Context Package (from Dual-Retrieval), and the RelationalContextVector.

It instantly generates delta_weights for the Momentary Modulation LoRA. These are subtle nudges to Samantha's personality for this specific turn.

The MomentaryLoRAStateManager applies the recurrence ΔW_Mod_t = (decay * A) @ ΔW_Mod_{t-1} + B @ δ_t. This ensures the momentary adaptation has continuity from the previous turn, preventing jarring shifts. The new Momentary Modulation LoRA is hot-stored in Redis.

The core LLM (8B DeepSeekR1/Qwen3, running on vLLM/LoRAX with 4-bit quantization on your A100/H100) now performs inference. Crucially, it loads and applies TWO LoRAs simultaneously:

The static Base Personality LoRA (pre-trained offline, loaded once at startup from PostgreSQL) defines Samantha's consistent core identity (her inherent empathy, curiosity, voice).

The dynamically generated Momentary Modulation LoRA (from the current turn's HyperNetwork output) subtly adapts her core personality for this specific interaction (e.g., a hint of warmth, a touch of playfulness).

The LLM generates a draft_response.

The CoherenceGuardian (NLI model) checks the draft_response against Samantha's recent statements and any initial user beliefs in the BeliefGraph to ensure consistency. If incoherent, it triggers a regeneration with a constraint.

The final response text is sent to the TTSSystem for audio output.

What's Happening in the Background (Asynchronous & Persistent):

Async State Persistor: Every few seconds (e.g., 10s), this background task wakes up. It reads the current Momentary Modulation LoRA, RelationalContextVector, and UserModel snapshots from Redis and asynchronously writes them to PostgreSQL. This ensures resilience; if your system crashes, Samantha will pick up almost exactly where she left off.

Autonomous Reflection Engine: After enough interactions (e.g., first reflection cycle after a few hours), this engine starts. It samples interactions from your interaction_log and prompts the AgenticCore's LLM to find "novel, non-obvious connections or underlying patterns." These generated "insights" are then added to your BeliefGraph in PostgreSQL, becoming "her own thoughts" about you and your relationship.
