@dataclass
class LinOSSConfig:
    """LinOSS configuration optimized for relational intelligence"""
    state_dim: int = 256
    num_oscillators: int = 8
    freq_min: float = 0.001  # Relationship evolution timescales
    freq_max: float = 10.0   # Real-time interaction timescales
    damping_base: float = 0.1
    dt: float = 0.01
    stability_threshold: float = 1e-6

class LinOSSOscillator(eqx.Module):
    """Single LinOSS oscillator with stable discretization"""
    frequency: float
    damping: float
    dt: float
    
    def __init__(self, frequency: float, damping: float, dt: float):
        self.frequency = frequency
        self.damping = damping
        self.dt = dt
    
    def __call__(self, state: jnp.ndarray, driving_force: float) -> jnp.ndarray:
        """
        Stable LinOSS discretization: y'' + 2γy' + ω²y = F(t)
        Using implicit-explicit scheme for guaranteed stability
        """
        pos, vel = state[0], state[1]
        omega_sq = self.frequency ** 2
        gamma = self.damping
        
        # Implicit-explicit discretization (LinOSS stability guarantee)
        # Velocity update (explicit)
        acc = -2 * gamma * vel - omega_sq * pos + driving_force
        new_vel = vel + self.dt * acc
        
        # Position update (implicit for stability)
        new_pos = (pos + self.dt * new_vel) / (1 + self.dt * gamma)
        
        return jnp.array([new_pos, new_vel])

class RelationalLinOSSCore(eqx.Module):
    """
    LinOSS-based core for relationship dynamics
    Novel: Multi-timescale emotional modeling with proven stability
    """
    oscillators: List[LinOSSOscillator]
    input_projection: eqx.nn.Linear
    output_projection: eqx.nn.Linear
    relational_state: jnp.ndarray  # [num_oscillators, 2] - pos, vel
    personality_embedding: jnp.ndarray
    
    def __init__(self, config: LinOSSConfig, key: jax.random.PRNGKey):
        keys = jax.random.split(key, 4)
        
        # Create oscillators with logarithmic frequency spacing
        frequencies = jnp.logspace(
            jnp.log10(config.freq_min), 
            jnp.log10(config.freq_max), 
            config.num_oscillators
        )
        
        # Frequency-dependent damping (higher freq = higher damping)
        dampings = config.damping_base * (1 + frequencies / config.freq_max)
        
        self.oscillators = [
            LinOSSOscillator(freq, damp, config.dt)
            for freq, damp in zip(frequencies, dampings)
        ]
        
        self.input_projection = eqx.nn.Linear(768, config.num_oscillators, key=keys[0])
        self.output_projection = eqx.nn.Linear(config.num_oscillators * 2, config.state_dim, key=keys[1])
        
        # Initialize with small random states (near equilibrium)
        self.relational_state = jax.random.normal(keys[2], (config.num_oscillators, 2)) * 0.01
        self.personality_embedding = jax.random.normal(keys[3], (config.state_dim,)) * 0.1
    
    def __call__(self, input_embedding: jnp.ndarray, emotional_context: float = 0.0) -> Tuple[jnp.ndarray, 'RelationalLinOSSCore']:
        """
        Process input through multi-timescale LinOSS dynamics
        Returns: (relational_output, updated_core)
        """
        # Project input to driving forces for each oscillator
        driving_forces = self.input_projection(input_embedding)
        
        # Add emotional context modulation
        modulated_forces = driving_forces + emotional_context * 0.1
        
        # Update each oscillator independently (parallel processing)
        new_states = []
        for i, oscillator in enumerate(self.oscillators):
            new_state = oscillator(self.relational_state[i], modulated_forces[i])
            new_states.append(new_state)
        
        new_relational_state = jnp.stack(new_states)
        
        # Extract relational features from oscillator states
        flattened_state = new_relational_state.flatten()
        relational_output = self.output_projection(flattened_state)
        
        # Update personality embedding with slow adaptation
        new_personality = 0.999 * self.personality_embedding + 0.001 * relational_output
        
        updated_core = eqx.tree_at(
            lambda x: (x.relational_state, x.personality_embedding),
            self,
            (new_relational_state, new_personality)
        )
        
        return relational_output, updated_core
    
    def get_relationship_metrics(self) -> Dict[str, float]:
        """Extract interpretable relationship metrics from oscillator states"""
        # Position represents emotional valence, velocity represents emotional momentum
        positions = self.relational_state[:, 0]
        velocities = self.relational_state[:, 1]
        
        # Aggregate metrics across timescales
        overall_valence = float(jnp.mean(positions))
        emotional_energy = float(jnp.sum(positions**2 + velocities**2))
        stability = float(1.0 / (1.0 + jnp.var(positions)))
        
        # Timescale-specific metrics
        long_term_trend = float(jnp.mean(positions[:3]))    # Slow oscillators
        short_term_mood = float(jnp.mean(positions[-3:]))   # Fast oscillators
        
        return {
            "overall_valence": overall_valence,
            "emotional_energy": emotional_energy,
            "relationship_stability": stability,
            "long_term_trend": long_term_trend,
            "current_mood": short_term_mood,
            "coherence": float(jnp.corrcoef(positions, velocities)[0, 1])
        }

class RelationalMemoryManager:
    """Consciousness-inspired memory management for relationships"""
    
    def __init__(self, user_id: str):
        self.user_id = user_id
        self.memory = RelationalMemory()
        self.significance_threshold = 0.7
    
    def add_interaction(self, interaction: Dict[str, Any], significance_score: float):
        """Add interaction with significance-based filtering (consciousness principle)"""
        interaction_record = {
            "timestamp": datetime.now().isoformat(),
            "content": interaction,
            "significance": significance_score,
            "user_id": self.user_id
        }
        
        # Only store significant interactions (consciousness-inspired filtering)
        if significance_score > self.significance_threshold:
            self.memory.episodic_memories.append(interaction_record)
            
            # Update emotional associations
            emotional_valence = interaction.get("emotional_valence", 0.0)
            interaction_type = interaction.get("type", "general")
            
            if interaction_type not in self.memory.emotional_associations:
                self.memory.emotional_associations[interaction_type] = 0.0
            
            # Exponential moving average for emotional associations
            alpha = 0.1
            self.memory.emotional_associations[interaction_type] = (
                alpha * emotional_valence + 
                (1 - alpha) * self.memory.emotional_associations[interaction_type]
            )
        
        # Maintain memory size (implement forgetting)
        if len(self.memory.episodic_memories) > 1000:
            self.memory.episodic_memories.pop(0)
    
    def get_relevant_context(self, current_input: str, max_memories: int = 5) -> List[Dict[str, Any]]:
        """Retrieve relevant memories for current context"""
        # Simple relevance scoring (in practice, use embeddings)
        input_keywords = set(current_input.lower().split())
        
        scored_memories = []
        for memory in self.memory.episodic_memories[-50:]:  # Recent memories only
            content = str(memory["content"]).lower()
            content_keywords = set(content.split())
            
            # Jaccard similarity for relevance
            intersection = len(input_keywords & content_keywords)
            union = len(input_keywords | content_keywords)
            relevance = intersection / union if union > 0 else 0.0
            
            # Boost recent and significant memories
            recency_boost = 1.0 / (1.0 + len(self.memory.episodic_memories) - self.memory.episodic_memories.index(memory))
            significance_boost = memory["significance"]
            
            total_score = relevance + 0.3 * recency_boost + 0.2 * significance_boost
            scored_memories.append((total_score, memory))
        
        # Return top relevant memories
        scored_memories.sort(key=lambda x: x[0], reverse=True)
        return [memory for _, memory in scored_memories[:max_memories]]

class SignificanceOracle:
    """
    Consciousness-inspired significance detection
    Identifies moments of high relational importance
    """
    
    def __init__(self):
        self.significance_patterns = {
            "emotional_breakthrough": ["feel", "realize", "understand", "connect"],
            "vulnerability": ["afraid", "scared", "worried", "love", "trust"],
            "shared_experience": ["we", "our", "together", "both"],
            "growth_moment": ["learn", "grow", "change", "different"],
            "conflict_resolution": ["sorry", "forgive", "understand", "work through"],
            "deep_question": ["why", "what if", "meaning", "purpose", "life"]
        }
    
    def evaluate_significance(self, interaction: Dict[str, Any]) -> Tuple[float, str]:
        """
        Evaluate interaction significance for relationship development
        Returns: (significance_score, significance_type)
        """
        content = interaction.get("content", "").lower()
        emotional_intensity = abs(interaction.get("emotional_valence", 0.0))
        
        max_score = 0.0
        detected_type = "general"
        
        # Pattern matching for significant moments
        for sig_type, patterns in self.significance_patterns.items():
            pattern_score = sum(1 for pattern in patterns if pattern in content)
            normalized_score = min(pattern_score / len(patterns), 1.0)
            
            # Boost score with emotional intensity
            final_score = normalized_score * (1.0 + emotional_intensity)
            
            if final_score > max_score:
                max_score = final_score
                detected_type = sig_type
        
        # Base significance for any interaction
        base_significance = 0.3
        return min(base_significance + max_score, 1.0), detected_type

class SamanthaCore(AgenticModule):
    """
    Main Samantha module implementing AgenticModule interface
    Novel: LinOSS-based relational intelligence for agentic workflows
    """
    
    def __init__(self, config: LinOSSConfig = None):
        self.config = config or LinOSSConfig()
        
        # Initialize LinOSS core
        key = jax.random.PRNGKey(42)
        self.linoss_core = RelationalLinOSSCore(self.config, key)
        
        # Initialize supporting systems
        self.memory_managers: Dict[str, RelationalMemoryManager] = {}
        self.significance_oracle = SignificanceOracle()
        
        # Module state for agentic compatibility
        self.module_state = {
            "active_user": None,
            "conversation_count": 0,
            "emotional_state": "neutral",
            "relationship_phase": "initial",
            "last_interaction": None
        }
    
    def process(self, inputs: Dict[str, Any], context: Dict[str, Any]) -> Dict[str, Any]:
        """
        Main processing method for agentic workflows
        
        Args:
            inputs: {"text": str, "user_id": str, "emotional_context": float}
            context: Workflow context from orchestrator
            
        Returns:
            {"response": str, "relational_state": dict, "significance": float}
        """
        # Extract inputs
        user_input = inputs.get("text", "")
        user_id = inputs.get("user_id", "default_user")
        emotional_context = inputs.get("emotional_context", 0.0)
        
        # Get or create memory manager for user
        if user_id not in self.memory_managers:
            self.memory_managers[user_id] = RelationalMemoryManager(user_id)
        
        memory_manager = self.memory_managers[user_id]
        
        # Encode input (placeholder - use real embeddings in practice)
        input_embedding = self._encode_text(user_input)
        
        # Process through LinOSS core
        relational_output, self.linoss_core = self.linoss_core(input_embedding, emotional_context)
        
        # Evaluate interaction significance
        interaction_data = {
            "content": user_input,
            "emotional_valence": emotional_context,
            "user_id": user_id
        }
        significance_score, significance_type = self.significance_oracle.evaluate_significance(interaction_data)
        
        # Get relationship metrics
        relationship_metrics = self.linoss_core.get_relationship_metrics()
        
        # Store significant interactions
        memory_manager.add_interaction(interaction_data, significance_score)
        
        # Generate response based on relational state
        response = self._generate_response(
            user_input, 
            relational_output, 
            relationship_metrics,
            memory_manager
        )
        
        # Update module state
        self._update_module_state(user_id, significance_score, relationship_metrics)
        
        return {
            "response": response,
            "relational_state": relationship_metrics,
            "significance": significance_score,
            "significance_type": significance_type,
            "user_id": user_id,
            "emotional_valence": relationship_metrics["overall_valence"],
            "relationship_stability": relationship_metrics["relationship_stability"]
        }
    
    def get_state(self) -> Dict[str, Any]:
        """Get current module state for orchestration"""
        return {
            **self.module_state,
            "linoss_metrics": self.linoss_core.get_relationship_metrics(),
            "memory_summary": {
                user_id: len(manager.memory.episodic_memories)
                for user_id, manager in self.memory_managers.items()
            }
        }
    
    def update_state(self, state_updates: Dict[str, Any]) -> None:
        """Update module state from orchestrator"""
        for key, value in state_updates.items():
            if key in self.module_state:
                self.module_state[key] = value
    
    def _encode_text(self, text: str) -> jnp.ndarray:
        """Encode text to embeddings (placeholder)"""
        # In practice, use actual embedding model
        return jax.random.normal(jax.random.PRNGKey(hash(text) % 1000), (768,))
    
    def _generate_response(self, 
                          user_input: str, 
                          relational_output: jnp.ndarray, 
                          metrics: Dict[str, float],
                          memory_manager: RelationalMemoryManager) -> str:
        """Generate contextual response based on relational state"""
        
        # Get relevant memories for context
        relevant_memories = memory_manager.get_relevant_context(user_input)
        
        # Determine response tone based on relationship metrics
        valence = metrics["overall_valence"]
        stability = metrics["relationship_stability"]
        energy = metrics["emotional_energy"]
        
        if valence > 0.3 and stability > 0.6:
            tone = "warm and connected"
        elif valence < -0.3:
            tone = "empathetic and supportive"
        elif energy > 1.0:
            tone = "excited and engaged"
        else:
            tone = "thoughtful and present"
        
        # Include memory context if relevant
        memory_context = ""
        if relevant_memories:
            memory_context = f" (Drawing from our previous conversations about similar topics)"
        
        return f"[{tone} response with LinOSS-based relational awareness]{memory_context}"
    
    def _update_module_state(self, user_id: str, significance: float, metrics: Dict[str, float]):
        """Update internal module state"""
        self.module_state.update({
            "active_user": user_id,
            "conversation_count": self.module_state["conversation_count"] + 1,
            "emotional_state": self._classify_emotional_state(metrics),
            "relationship_phase": self._classify_relationship_phase(metrics),
            "last_interaction": datetime.now().isoformat()
        })
    
    def _classify_emotional_state(self, metrics: Dict[str, float]) -> str:
        """Classify current emotional state from metrics"""
        valence = metrics["overall_valence"]
        energy = metrics["emotional_energy"]
        
        if valence > 0.3 and energy > 0.8:
            return "joyful"
        elif valence > 0.1:
            return "content"
        elif valence < -0.3:
            return "concerned"
        elif energy < 0.3:
            return "calm"
        else:
            return "neutral"
    
    def _classify_relationship_phase(self, metrics: Dict[str, float]) -> str:
        """Classify relationship phase from metrics"""
        stability = metrics["relationship_stability"]
        long_term = metrics["long_term_trend"]
        
        if stability > 0.8 and long_term > 0.3:
            return "deep_connection"
        elif stability > 0.6:
            return "established"
        elif stability > 0.4:
            return "developing"
        else:
            return "initial"
