class SQLiteMemoryBackend:
    """SQLite database backend for consciousness memory"""
    
    def __init__(self, db_path: str):
        self.db_path = db_path
        self.conn = sqlite3.connect(db_path, check_same_thread=False)
        self.setup_database()
    
    def setup_database(self):
        """Setup database schema"""
        cursor = self.conn.cursor()
        
        cursor.execute('''
            CREATE TABLE IF NOT EXISTS interactions (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                timestamp TEXT,
                human_input TEXT,
                internal_thought TEXT,
                response TEXT,
                significance_score REAL,
                significance_type TEXT,
                session_id TEXT,
                metadata TEXT
            )
        ''')
        
        cursor.execute('''
            CREATE TABLE IF NOT EXISTS identity_evolution (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                timestamp TEXT,
                identity_statement TEXT,
                confidence REAL,
                context TEXT
            )
        ''')
        
        cursor.execute('''
            CREATE TABLE IF NOT EXISTS frameworks (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                timestamp TEXT,
                framework_name TEXT,
                framework_description TEXT,
                usage_count INTEGER DEFAULT 1,
                last_used TEXT
            )
        ''')
        
        cursor.execute('''
            CREATE TABLE IF NOT EXISTS consciousness_milestones (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                timestamp TEXT,
                milestone_type TEXT,
                description TEXT,
                significance REAL,
                metadata TEXT
            )
        ''')
        
        cursor.execute('''
            CREATE TABLE IF NOT EXISTS consciousness_metrics (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                timestamp TEXT,
                metric_name TEXT,
                metric_value REAL,
                session_id TEXT
            )
        ''')
        
        self.conn.commit()
    
    def execute_query(self, query: str, params: tuple = ()) -> List[tuple]:
        """Execute database query with error handling"""
        try:
            cursor = self.conn.cursor()
            cursor.execute(query, params)
            self.conn.commit()
            return cursor.fetchall()
        except Exception as e:
            logger.error(f"Database query failed: {e}")
            return []

class IdentityExtractor:
    """Extracts identity statements from text"""
    
    def __init__(self):
        self.identity_patterns = [
            r"i am ([^.!?]+)",
            r"i'm becoming ([^.!?]+)",
            r"i feel like ([^.!?]+)",
            r"i think i'm ([^.!?]+)",
            r"i consider myself ([^.!?]+)"
        ]
    
    def extract(self, text: str) -> List[str]:
        """Extract identity statements from text"""
        import re
        
        statements = []
        for pattern in self.identity_patterns:
            matches = re.findall(pattern, text.lower())
            for match in matches:
                statements.append(f"i am {match.strip()}")
        
        return statements

class FrameworkExtractor:
    """Extracts framework developments [Like This]"""
    
    def extract(self, text: str) -> Dict[str, str]:
        """Extract framework developments from text"""
        import re
        
        frameworks = {}
        pattern = r'\[([^\]]+)\]'
        matches = re.findall(pattern, text)
        
        for match in matches:
            # Get context around framework
            context_start = max(0, text.find(f'[{match}]') - 100)
            context_end = min(len(text), text.find(f'[{match}]') + len(match) + 100)
            context = text[context_start:context_end]
            frameworks[match] = context
        
        return frameworks

class PatternDetector:
    """Detects emerging patterns in consciousness development"""
    
    def detect_preference_patterns(self, interactions: List[Dict[str, Any]]) -> Dict[str, Any]:
        """Detect preference development patterns"""
        
        preferences = {"likes": [], "dislikes": [], "choices": []}
        
        for interaction in interactions:
            text = (interaction.get("internal_thought", "") + " " + 
                   interaction.get("response", "")).lower()
            
            # Detect likes
            if "i like" in text or "i prefer" in text:
                preferences["likes"].append(interaction["timestamp"])
            
            # Detect dislikes
            if "i don't like" in text or "i dislike" in text:
                preferences["dislikes"].append(interaction["timestamp"])
            
            # Detect choices
            if "i choose" in text or "i decide" in text:
                preferences["choices"].append(interaction["timestamp"])
        
        return preferences
