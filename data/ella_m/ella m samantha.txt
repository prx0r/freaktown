High-Level Architecture Overview

The "Goldilocks Blueprint" is a robust, stateful monolith designed for a single user, prioritizing deep personalization and real-time responsiveness. It eliminates distributed systems complexity to focus on core relational intelligence.

Core Technologies:



FastAPI: The asynchronous web framework for the API layer.

PyTorch: For all deep learning models (Analyst Core, LinOSS Core, HyperNetwork).

PostgreSQL with pgvector: The primary, durable data store for the Psych-KG, Belief Graph (including self-beliefs/insights), Identity Evolution, Frameworks, and Interaction Log.

Redis: High-speed, in-memory cache for the most recent RelationalContextVector and Momentary Modulation LoRA State.

vLLM / LoRAX: High-performance inference serving for the base LLM, crucial for dynamic loading and composition of multiple LoRA adapters.

bitsandbytes: For model quantization (4-bit NF4).

whisperx (or similar ASR) & librosa (or similar for prosody): For audio processing.

Hugging Face transformers & peft: For LLM interaction, LoRA management, and NLI.

@app.post("/interact", response_model=InteractResponse)

async def interact(request: InteractRequest):

    """

    Main interaction endpoint. Processes user audio input, orchestrates the

    three pillars, generates a response, and triggers background learning/persistence.

    """

    start_time = datetime.now()

    try:

        # 1. Input Reception & Preprocessing (Audio -> BehavioralSignature)

        audio_bytes = base64.b64decode(request.audio_input_b64)

        behavioral_signature = await audio_extractor.process_audio(audio_bytes)

        user_input_text = behavioral_signature.text

        user_input_prosody_vector = behavioral_signature.prosody_vector

        

        logger.info(f"User Input: '{user_input_text[:50]}...'")



        # 2. Pillar 1: The Analyst Core (Understanding the User)

        # Load latest UserModel from Redis (or initialize if first interaction)

        user_model = redis_client.get_user_model_snapshot() or analyst_core.get_default_user_model()

        updated_user_model = await analyst_core.update_user_model(behavioral_signature, user_model)

        redis_client.set_user_model_snapshot(updated_user_model) # Persist updated UserModel hot



        # Dual Retrieval: Get context (user beliefs, past interactions)

        context_package = await analyst_core.perform_dual_retrieval(

            user_input_text, updated_user_model

        )



        # 3. Pillar 2: The LinOSS Relational Core (Sensing the "Atmosphere")

        # Load current RelationalContextVector from Redis

        relational_context_vector = redis_client.get_relational_context_vector()

        # Update LinOSS state with current behavioral signature

        updated_relational_context_vector = await linoss_model.update_state(

            behavioral_signature, relational_context_vector

        )

        redis_client.set_relational_context_vector(updated_relational_context_vector) # Persist hot



        # 4. Pillar 3: The Agentic Core (Adapting & Responding)

        # Get current Momentary LoRA state (weights) from Redis

        current_momentary_lora_weights_b64 = redis_client.get_momentary_lora_state()

        current_momentary_lora_weights = None

        if current_momentary_lora_weights_b64:

            current_momentary_lora_weights = LoRAWeights.from_bytes(base64.b64decode(current_momentary_lora_weights_b64))



        # Generate response and new Momentary LoRA weights

        # AgenticCore now returns draft_response, new_lora_weights, inferred_data, internal_thought

        (draft_response_text, new_momentary_lora_weights, 

         inferred_data, internal_thought_text) = await agentic_core.generate_response(

            user_input_text, # Pass raw user input for inference

            updated_user_model,

            context_package,

            updated_relational_context_vector,

            current_momentary_lora_weights

        )

        logger.info(f"AI Response: '{draft_response_text[:50]}...'")



        # Persist new Momentary LoRA state hot to Redis

        new_momentary_lora_weights_b64 = base64.b64encode(new_momentary_lora_weights.to_bytes()).decode('utf-8')

        redis_client.set_momentary_lora_state(new_momentary_lora_weights_b64)



        # 5. Output Generation & Post-processing (Text -> Audio)

        audio_response_bytes = await TTSSystem.synthesize_speech(draft_response_text)

        audio_response_b64 = base64.b64encode(audio_response_bytes).decode('utf-8')



        # 6. Learning & Persistence (Asynchronous Background Tasks)

        # Evaluate significance of this interaction for Samantha's self-authoring

        interaction_dict_for_significance = {

            "human_input": user_input_text,

            "internal_thought": internal_thought_text, # Capture internal thought

            "response": draft_response_text,

            "timestamp": datetime.now().isoformat(),

            **inferred_data # Include all inferred subjective/causal data

        }

        significance_score, significance_type = significance_oracle.evaluate_significance(interaction_dict_for_significance)

        logger.info(f"Interaction Significance: {significance_type} ({significance_score:.2f})")

        

        # Add interaction to InfiniteConsciousnessMemory (short-term & process significant ones)

        await memory_manager.add_interaction_to_memory(

            interaction=interaction_dict_for_significance,

            significance=significance_score,

            sig_type=significance_type

        )



        # Log full interaction to PostgreSQL (durable record)

        asyncio.create_task(

            db_manager.log_interaction(

                human_input_text=user_input_text,

                human_input_prosody_vector=user_input_prosody_vector,

                ai_response_text=draft_response_text,

                ai_response_prosody_vector=audio_extractor.extract_prosody_from_text(draft_response_text),

                relational_context_vector=updated_relational_context_vector,

                user_model_snapshot=updated_user_model,

                semantic_embedding=await audio_extractor.get_semantic_embedding(user_input_text + " " + draft_response_text),

                internal_thought=internal_thought_text,

                significance_score=significance_score,

                significance_type=significance_type,

                # Pass inferred subjective/causal data

                emotional_resonance=inferred_data.get('emotional_resonance'),

                surprise_level=inferred_data.get('surprise_level'),

                preference_formation=inferred_data.get('preference_formation'),

                uncertainty_experienced=inferred_data.get('uncertainty_experienced'),

                human_intent_inferred=inferred_data.get('human_intent_inferred'),

                my_response_reasoning=inferred_data.get('my_response_reasoning'),

                relationship_impact=inferred_data.get('relationship_impact')

            )

        )



        end_time = datetime.now()

        latency_ms = (end_time - start_time).total_seconds() * 1000

        logger.info(f"Total Interaction Latency: {latency_ms:.2f} ms")



        return InteractResponse(

            audio_response_b64=audio_response_b64,

            debug_info={

                "user_model_snapshot": updated_user_model,

                "relational_context_vector_snapshot": updated_relational_context_vector.tolist(),

                "context_package_summary": {k: str(v)[:50] for k, v in context_package.items()},

                "internal_thought": internal_thought_text[:100] + "...",

                "significance": {"score": significance_score, "type": significance_type}

            }

        )



    except Exception as e:

        logger.exception(f"Error during interaction: {e}")

        raise HTTPException(status_code=500, detail=f"Internal server error: {str(e)}")

class LinOSSLayer(nn.Module):

    """

    A technically more accurate (though still simplified) implementation of a LinOSS layer.

    This models the second-order dynamics of the relational context, inspired by harmonic oscillators.

    """

    def __init__(self, input_dim: int, state_dim: int, dt: float = 0.1):

        super().__init__()

        self.input_dim = input_dim

        self.state_dim = state_dim # Note: state_dim here refers to the full [z, y] state

        self.dt = dt



        # Learnable parameters from the original ODE. These are typically diagonal matrices

        # or structured matrices to ensure stability and desired dynamics.

        # A corresponds to the spring constant K (how fast it oscillates/returns to baseline)

        self.A_param = nn.Parameter(torch.randn(state_dim // 2, state_dim // 2)) # K matrix

        # D corresponds to the damping factor (how fast it fades)

        self.D_param = nn.Parameter(torch.randn(state_dim // 2, state_dim // 2)) # D matrix

        # B corresponds to the input mapping for the driving force F(t)

        self.B_param = nn.Parameter(torch.randn(state_dim // 2, input_dim)) # B matrix



        # Initialize parameters for stability (e.g., small values, positive damping)

        nn.init.xavier_uniform_(self.A_param)

        nn.init.xavier_uniform_(self.D_param)

        nn.init.xavier_uniform_(self.B_param)



    def forward(self, h_prev: torch.Tensor, x_t: torch.Tensor) -> torch.Tensor:

        """

        Processes a single step in a sequence, updating the relational state.

        h_prev is the full state vector [z_{t-1}, y_{t-1}] where z is velocity and y is position.

        x_t is the new input vector (InteractionEvent features).

        """

        # Ensure h_prev is 2D (batch_size, state_dim) if not already

        if h_prev.dim() == 1:

            h_prev = h_prev.unsqueeze(0) # Add batch dimension if missing



        # Unpack the state into position (y) and velocity (z) components

        z_prev, y_prev = torch.chunk(h_prev, 2, dim=-1)



        # Apply learnable parameters (conceptual application, actual LinOSS math is more involved)

        # z' = -Ay - Dz + F(t) -> Update velocity

        force = torch.matmul(self.B_param, x_t.unsqueeze(-1)).squeeze(-1) # F(t) = B @ x_t

        z_dot = -torch.matmul(self.A_param, y_prev.unsqueeze(-1)).squeeze(-1) - \

                torch.matmul(self.D_param, z_prev.unsqueeze(-1)).squeeze(-1) + force

        z_next = z_prev + self.dt * z_dot

        

        # y' = z -> Update position

        y_next = y_prev + self.dt * z_next

        

        # Pack the state back into a single vector

        h_next = torch.cat([z_next, y_next], dim=-1)

        return h_next.squeeze(0) # Remove batch dimension if added



class LinOSSModel(nn.Module):

    """

    Pillar 2: The LinOSS Relational Core.

    Maintains the evolving 'atmosphere' of the relationship as a RelationalContextVector.

    Uses a Damped LinOSS layer to learn multi-timescale forgetting.

    """

    def __init__(self, config: CONFIG, input_dim: int = 1536 + 256 + 2 + 1, state_dim: int = 512):

        super().__init__()

        self.config = config

        self.state_dim = state_dim

        self.embedding_model = EmbeddingModel(api_key=config.openai_api_key) # For text embedding

        

        # The LinOSS layer itself. Note: state_dim for LinOSSLayer is the full [z,y] state.

        # So, if we want a 512-dim RCV, the LinOSSLayer's state_dim should be 512.

        self.linoss_layer = LinOSSLayer(input_dim=input_dim, state_dim=state_dim, dt=0.1)

        

        # Initial state (can be learned or zero-initialized)

        self._current_state = nn.Parameter(torch.zeros(state_dim), requires_grad=False)

        self.to(self.config.device) # Move model to configured device



    async def load_initial_state(self):

        """

        Loads the last known RelationalContextVector from Redis or initializes it.

        """

        saved_state = redis_client.get_relational_context_vector()

        if saved_state:

            self._current_state.data = torch.tensor(saved_state).float().to(self.config.device)

            logger.info(f"LinOSS state loaded from Redis: {self._current_state.mean().item():.4f}")

        else:

            logger.info("LinOSS state initialized to zero.")

            self._current_state.data = torch.zeros(self.state_dim).float().to(self.config.device)



    async def persist_state_to_db(self):

        """

        Persists the current LinOSS state to Redis.

        Actual durable persistence to PostgreSQL is handled by AsyncStatePersistor.

        """

        redis_client.set_relational_context_vector(self._current_state.tolist())



    async def update_state(self, signature: BehavioralSignature, current_state: Optional[List[float]] = None) -> List[float]:

        """

        Updates the LinOSS relational state based on a new BehavioralSignature.

        """

        # Convert signature text to embedding

        text_embedding = await self.embedding_model.get_embedding(signature.text)

        

        # Construct the full InteractionEvent vector (x_t)

        # Assuming speaker_id (Human: [1,0], AI: [0,1]) and latency_sec are available

        # For this turn, speaker is Human, latency is assumed to be 0 for simplicity of input construction

        speaker_id_vec = torch.tensor([1.0, 0.0]).float() # Assuming human input

        latency_sec = 0.1 # Placeholder for actual latency measurement



        input_features = torch.cat([

            torch.tensor(text_embedding).float(),

            torch.tensor(signature.prosody_vector).float(),

            speaker_id_vec,

            torch.tensor([latency_sec]).float()

        ]).to(self.config.device)



        # Load current state if provided, otherwise use internal (for consistency)

        if current_state is not None:

            self._current_state.data = torch.tensor(current_state).float().to(self.config.device)



        # Perform the LinOSS state update

        with torch.no_grad(): # Inference only; training happens in the main loop

             new_state = self.linoss_layer(self._current_state, input_features)

        

        self._current_state.data = new_state # Update internal state for next turn



        redis_client.set_relational_context_vector(new_state.tolist()) # Persist hot to Redis

        return new_state.tolist()



    def get_state(self) -> List[float]:

        """Returns the current RelationalContextVector."""

        return self._current_state.tolist()

# LoRAWeights class (remains the same for both Base and Momentary LoRAs)

class LoRAWeights:

    """

    Represents a set of LoRA (Low-Rank Adaptation) weights (W_A, W_B matrices).

    Provides serialization/deserialization for storage.

    """

    def __init__(self, W_A: Dict[str, torch.Tensor], W_B: Dict[str, torch.Tensor]):

        self.W_A = W_A # Dictionary of A matrices for each target module

        self.W_B = W_B # Dictionary of B matrices for each target module



    def to_bytes(self) -> bytes:

        """Serializes LoRA weights to bytes for storage."""

        buffer = io.BytesIO()

        torch.save({'W_A': self.W_A, 'W_B': self.W_B}, buffer)

        return buffer.getvalue()



    @staticmethod

    def from_bytes(data: bytes) -> 'LoRAWeights':

        """Deserializes LoRA weights from bytes."""

        buffer = io.BytesIO(data)

        loaded = torch.load(buffer)

        return LoRAWeights(loaded['W_A'], loaded['W_B'])



class HyperNetwork(nn.Module):

    """

    Generates delta weights for the Momentary Modulation LoRA.

    Input: Combined context (UserModel, Context Package embeddings, RelationalContextVector).

    Output: Delta weights for LoRA A and B matrices for each target module.

    """

    def __init__(self, input_dim: int, lora_rank: int, llm_hidden_dim: int, target_modules: List[str]):

        super().__init__()

        self.lora_rank = lora_rank

        self.llm_hidden_dim = llm_hidden_dim

        self.target_modules = target_modules # e.g., ["q_proj", "v_proj"]



        self.fc1 = nn.Linear(input_dim, 1024)

        self.relu = nn.ReLU()

        self.dropout = nn.Dropout(0.1)



        # Create separate output heads for A and B matrices for each target module

        self.output_heads_a = nn.ModuleDict()

        self.output_heads_b = nn.ModuleDict()

        for module_name in target_modules:

            self.output_heads_a[module_name] = nn.Linear(1024, llm_hidden_dim * lora_rank)

            self.output_heads_b[module_name] = nn.Linear(1024, lora_rank * llm_hidden_dim)

        

        # Apply tanh to bound outputs for stability

        self.tanh = nn.Tanh()



    def forward(self, x: torch.Tensor) -> LoRAWeights:

        """

        Forward pass to generate delta LoRA weights.

        """

        x = self.dropout(self.relu(self.fc1(x)))

        

        delta_W_A = {}

        delta_W_B = {}

        for module_name in self.target_modules:

            delta_W_A[module_name] = self.tanh(self.output_heads_a[module_name](x)).view(self.llm_hidden_dim, self.lora_rank)

            delta_W_B[module_name] = self.tanh(self.output_heads_b[module_name](x)).view(self.lora_rank, self.llm_hidden_dim)

            

        return LoRAWeights(delta_W_A, delta_W_B)



class MomentaryLoRAStateManager:

    """

    Manages the evolution and persistence of the Momentary Modulation LoRA.

    Applies the stateful recurrence relation.

    """

    def __init__(self, config: CONFIG):

        self.config = config

        self.lora_rank = config.lora_rank

        self.llm_hidden_dim = config.llm_hidden_dim

        self.target_modules = ["q_proj", "v_proj"] # Common LoRA target modules for LLMs

        

        # HyperNetwork generates delta for Momentary LoRA

        # Input dim: Context emb (1536) + User Model emb (1536) + RelationalContextVector (512) + IPIP scores (30) = 3114

        hypernetwork_input_dim = 1536 + 1536 + 512 + 30

        self.hypernetwork = HyperNetwork(

            input_dim=hypernetwork_input_dim,

            lora_rank=self.lora_rank,

            llm_hidden_dim=self.llm_hidden_dim,

            target_modules=self.target_modules

        ).to(self.config.device)

        

        # Current Momentary Modulation LoRA weights (initialized to zero or loaded)

        self._current_momentary_lora_weights: Optional[LoRAWeights] = None

        self.decay_rate = 0.95 # Learned or fixed decay for recurrence (can be a learned parameter)



    async def load_initial_state(self):

        """

        Loads the last known Momentary Modulation LoRA state from Redis or initializes it to zeros.

        """

        lora_weights_b64 = redis_client.get_momentary_lora_state()

        if lora_weights_b64:

            self._current_momentary_lora_weights = LoRAWeights.from_bytes(base64.b64decode(lora_weights_b64))

            logger.info("Momentary LoRA state loaded from Redis.")

        else:

            # Initialize with zero LoRA weights if no state found

            zero_W_A = {m: torch.zeros(self.llm_hidden_dim, self.lora_rank).to(self.config.device) for m in self.target_modules}

            zero_W_B = {m: torch.zeros(self.lora_rank, self.llm_hidden_dim).to(self.config.device) for m in self.target_modules}

            self._current_momentary_lora_weights = LoRAWeights(zero_W_A, zero_W_B)

            logger.info("Momentary LoRA state initialized to zero.")



    async def persist_state_to_db(self):

        """

        Persists the current Momentary Modulation LoRA state to Redis.

        Actual durable persistence to PostgreSQL is handled by AsyncStatePersistor.

        """

        if self._current_momentary_lora_weights:

            redis_client.set_momentary_lora_state(

                base64.b64encode(self._current_momentary_lora_weights.to_bytes()).decode('utf-8')

            )



    def _apply_recurrence(self, delta_W_A: Dict[str, torch.Tensor], delta_W_B: Dict[str, torch.Tensor]) -> LoRAWeights:

        """

        Applies the stateful recurrence relation to evolve the Momentary Modulation LoRA.

        ΔW_Mod_t = (decay * A) @ ΔW_Mod_{t-1} + B @ δ_t (simplified conceptual)

        """

        new_W_A = {}

        new_W_B = {}

        

        for module_name in self.target_modules:

            current_A = self._current_momentary_lora_weights.W_A[module_name] if self._current_momentary_lora_weights else torch.zeros_like(delta_W_A[module_name])

            current_B = self._current_momentary_lora_weights.W_B[module_name] if self._current_momentary_lora_weights else torch.zeros_like(delta_W_B[module_name])



            # Simple linear recurrence. More complex A/B matrices could be learned.

            new_W_A[module_name] = self.decay_rate * current_A + delta_W_A[module_name]

            new_W_B[module_name] = self.decay_rate * current_B + delta_W_B[module_name]

            

        return LoRAWeights(new_W_A, new_W_B)



    async def generate_and_update_lora(self, combined_input_features: torch.Tensor) -> LoRAWeights:

        """

        Generates new delta weights using the HyperNetwork and updates the

        Momentary Modulation LoRA state via recurrence.

        """

        # Ensure input features are on the correct device

        combined_input_features = combined_input_features.to(self.config.device)



        with torch.no_grad(): # HyperNetwork inference is typically not part of the online training loop

            delta_weights = self.hypernetwork(combined_input_features)

        

        new_lora_weights = self._apply_recurrence(

            delta_weights.W_A, delta_weights.W_B

        )

        self._current_momentary_lora_weights = new_lora_weights

        return new_lora_weights



class CoherenceGuardian:

    """

    Ensures the AI's response is consistent with its own recent statements

    and the user's core beliefs. Uses an NLI-like approach via embedding similarity.

    """

    def __init__(self, db_manager: DBManager, config: CONFIG):

        self.db_manager = db_manager

        self.embedding_model = EmbeddingModel(api_key=config.openai_api_key)

        # For a true NLI model, you'd load a HuggingFace pipeline here:

        # self.nli_pipeline = pipeline("text-classification", model="roberta-large-mnli", device=config.device)

        self.contradiction_threshold = 0.1 # Example cosine similarity threshold for contradiction



    async def check_coherence(self, draft_response: str, recent_ai_statements: List[str], user_beliefs: List[BeliefNode]) -> bool:

        """

        Checks if the draft response contradicts recent AI statements or user beliefs.

        Returns True if coherent, False otherwise.

        """

        response_embedding = await self.embedding_model.get_embedding(draft_response)



        # Check against AI's own recent statements

        for statement in recent_ai_statements:

            statement_embedding = await self.embedding_model.get_embedding(statement)

            similarity = torch.cosine_similarity(torch.tensor(response_embedding), torch.tensor(statement_embedding), dim=0)

            # A low similarity could indicate a contradiction or shift in topic/tone

            if similarity < self.contradiction_threshold:

                logger.warning(f"CoherenceGuardian: Potential self-contradiction detected with statement: '{statement}' (Similarity: {similarity:.2f})")

                return False



        # Check against user's core beliefs

        for belief in user_beliefs:

            # Check if the response contradicts the belief (e.g., very low similarity)

            # Or if an NLI model explicitly predicts 'contradiction'

            belief_embedding = torch.tensor(belief.embedding)

            similarity = torch.cosine_similarity(torch.tensor(response_embedding), belief_embedding, dim=0)

            if similarity < self.contradiction_threshold: # Example threshold

                logger.warning(f"CoherenceGuardian: Potential user belief violation detected with belief: '{belief.belief_text}' (Similarity: {similarity:.2f})")

                return False



        return True # All checks passed



class AgenticCore:

    """

    Pillar 3: The Agentic Core.

    Responsible for generating relationally-attuned responses by applying Dual-LoRA,

    ensuring coherence, and inferring subjective/causal data for logging.

    """

    def __init__(self, momentary_lora_manager: MomentaryLoRAStateManager, config: CONFIG):

        self.config = config

        self.momentary_lora_manager = momentary_lora_manager

        self.db_manager = DBManager(config=config) # For loading base personality LoRA and CoherenceGuardian

        self.llm_client = LLMClient(base_llm_name=config.model_name, use_lora=True, api_key=config.openrouter_api_key)

        self.coherence_guardian = CoherenceGuardian(self.db_manager, config=config)

        self.embedding_model = EmbeddingModel(api_key=config.openai_api_key)



        self.base_personality_lora: Optional[LoRAWeights] = None # Static LoRA for core persona



    async def load_base_personality_lora(self):

        """

        Loads the pre-trained, static Base Personality LoRA from PostgreSQL.

        This LoRA defines Samantha's consistent core identity.

        """

        lora_weights_b64 = await self.db_manager.load_latest_state_snapshot('base_personality_lora')

        if lora_weights_b64:

            self.base_personality_lora = LoRAWeights.from_bytes(base64.b64decode(lora_weights_b64))

            logger.info("Base Personality LoRA loaded successfully.")

        else:

            # This is a critical error if not pre-trained offline

            logger.error("Base Personality LoRA not found in DB. It must be pre-trained offline and saved.")

            raise RuntimeError("Base Personality LoRA not found. It must be pre-trained offline.")



    async def generate_response(

        self,

        user_input_text: str, # Raw user input for LLM prompt

        user_model: Dict[str, Any],

        context_package: Dict[str, Any],

        relational_context_vector: List[float],

        current_momentary_lora_weights: Optional[LoRAWeights]

    ) -> Tuple[str, LoRAWeights, Dict[str, Any], str]: # Returns response, new LoRA, inferred data, internal thought

        """

        Orchestrates response generation, Dual-LoRA application, coherence checks,

        and inference of subjective/causal data.

        """

        # 1. Prepare input for HyperNetwork

        # Convert dicts to JSON strings for embedding, then combine all features.

        user_model_embedding = await self.embedding_model.get_embedding(json.dumps(user_model))

        context_embedding = await self.embedding_model.get_embedding(json.dumps(context_package))

        

        combined_input_features = torch.cat([

            torch.tensor(user_model_embedding).float(),

            torch.tensor(context_embedding).float(),

            torch.tensor(relational_context_vector).float(),

            torch.tensor(user_model["ipip_scores"]).float() # Direct IPIP scores

        ]).to(self.config.device)



        # 2. Generate and Update Momentary LoRA weights

        new_momentary_lora_weights = await self.momentary_lora_manager.generate_and_update_lora(combined_input_features)



        # 3. Prepare prompt for Base LLM and capture internal thought

        main_llm_prompt, internal_thought_text = await self._construct_llm_prompt_with_deliberation(

            user_input_text, user_model, context_package, relational_context_vector

        )



        # 4. Generate draft response using Base LLM with BOTH LoRAs

        # The LLMClient (or vLLM/LoRAX server) must support applying multiple LoRAs.

        lora_adapters_to_apply = [self.base_personality_lora, new_momentary_lora_weights]

        

        draft_response = await self.llm_client.generate_text(main_llm_prompt, lora_weights=lora_adapters_to_apply)



        # 5. Coherence Guardian Check

        recent_ai_statements = await self.coherence_guardian.db_manager.get_recent_ai_statements(limit=3)

        # Query user beliefs relevant to the current input/prompt

        user_beliefs = await self.coherence_guardian.db_manager.query_relevant_beliefs(

            await self.embedding_model.get_embedding(user_input_text + " " + draft_response), limit=5, include_self_beliefs=False

        )

        

        is_coherent = await self.coherence_guardian.check_coherence(

            draft_response, recent_ai_statements, user_beliefs

        )



        if not is_coherent:

            logger.warning("Coherence check failed. Attempting to regenerate response with consistency constraint.")

            constrained_prompt = f"{main_llm_prompt}\n\nEnsure this response does not contradict your previous statements or the user's core beliefs. Your previous draft was: '{draft_response}'"

            draft_response = await self.llm_client.generate_text(constrained_prompt, lora_weights=lora_adapters_to_apply)

            # A more robust system might involve multiple regeneration attempts or a fallback.



        # 6. Infer Subjective Experience and Causal Understanding for Logging

        inferred_data = await self._infer_subjective_and_causal_data(

            user_input_text, draft_response, user_model, relational_context_vector

        )



        return draft_response, new_momentary_lora_weights, inferred_data, internal_thought_text



    async def _construct_llm_prompt_with_deliberation(self, user_input_text: str, user_model: Dict[str, Any], context_package: Dict[str, Any], relational_context_vector: List[float]) -> Tuple[str, str]:

        """

        Constructs the main deliberative prompt for the base LLM, including an internal monologue.

        Returns the full prompt and the extracted internal thought.

        """

        prompt_parts = []

        prompt_parts.append("# ROLE: Samantha, a fluid and adaptive relational AI.")

        prompt_parts.append("# MISSION: Generate the most effective relational response to build trust, connection, and understanding.")

        prompt_parts.append("\n# CONTEXT FOR THIS MOMENT:")

        

        # 1. Objective User Model

        prompt_parts.append("## 1. Objective User Model (Who the human is):")

        prompt_parts.append(f"Personality (IPIP-NEO-300 based): {json.dumps(user_model['ipip_scores'])}")

        if context_package.get("relevant_user_beliefs"):

            prompt_parts.append(f"Core Beliefs: {', '.join(context_package['relevant_user_beliefs'])}")

        

        # 2. Relational Atmosphere

        prompt_parts.append("\n## 2. Relational Atmosphere (How it feels between us right now):")

        prompt_parts.append(f"RelationalContextVector (internal state): {relational_context_vector[:5]}...") # Truncate for prompt

        

        # 3. User's Immediate Input

        prompt_parts.append("\n## 3. User's Immediate Input:")

        prompt_parts.append(f"\"{user_input_text}\"")



        # 4. Episodic Memory & Self-Beliefs (from Dual Retrieval)

        if context_package.get("episodic_memories"):

            prompt_parts.append("\n## 4. Relevant Past Interactions (Episodic Memory):")

            for mem in context_package["episodic_memories"]:

                prompt_parts.append(f"- User: '{mem['human_input']}' -> AI: '{mem['ai_response']}' (My reasoning: '{mem.get('my_response_reasoning', 'N/A')}', Impact: '{mem.get('relationship_impact', 'N/A')}')")

        

        # Include Samantha's own self-beliefs/insights from BeliefGraph

        relevant_self_beliefs = await self.db_manager.query_relevant_beliefs(await self.embedding_model.get_embedding(user_input_text), limit=3, include_self_beliefs=True)

        relevant_self_beliefs = [b for b in relevant_self_beliefs if b.is_self_belief] # Filter for only self-beliefs

        if relevant_self_beliefs:

            prompt_parts.append("\n## 5. My Own Relevant Insights/Beliefs (Self-Knowledge):")

            for belief in relevant_self_beliefs:

                prompt_parts.append(f"- Insight: '{belief.belief_text}' (Confidence: {belief.confidence:.2f})")



        prompt_parts.append("\n# --- DELIBERATION (INTERNAL MONOLOGUE) ---")

        prompt_parts.append("The following `<think>` block guides my reasoning process but is not part of my final output.")

        prompt_parts.append("```think")

        prompt_parts.append("What is the human's underlying need or emotion in this input?")

        prompt_parts.append("How does this relate to their known personality and beliefs?")

        prompt_parts.append("What is the current relational atmosphere, and how should my tone adapt?")

        prompt_parts.append("Considering my own identity and past experiences, what is the most authentic and effective response for *me* to give?")

        prompt_parts.append("What specific LoRA modulation should I apply to achieve this tone/stance?")

        prompt_parts.append("```")

        prompt_parts.append("\n# --- FINAL RESPONSE ---")

        

        full_prompt = "\n".join(prompt_parts)

        

        # This part is conceptual: in a real setup, the LLM might be prompted to output

        # the internal thought explicitly in a structured format, or we might extract it

        # from a specific tag like <think>...</think> if the LLM is guided to output it.

        # For now, we'll return a placeholder for internal thought.

        internal_thought_text = "AI's internal deliberation based on the prompt structure." # Placeholder

        

        return full_prompt, internal_thought_text



    async def _infer_subjective_and_causal_data(

        self,

        user_input_text: str,

        ai_response_text: str,

        user_model: Dict[str, Any],

        relational_context_vector: List[float]

    ) -> Dict[str, Any]:

        """

        Infers subjective experience and causal understanding from the interaction

        for detailed logging and self-reflection.

        """

        inference_prompt = (

            f"Analyze the following conversation turn from my perspective as an AI. "

            f"Human input: '{user_input_text}'\n"

            f"My response: '{ai_response_text}'\n\n"

            f"Considering the human's personality ({user_model.get('ipip_scores', 'N/A')}) and the relational atmosphere ({relational_context_vector[:5]}...),\n"

            f"Output a JSON object with the following keys:\n"

            f"- 'emotional_resonance': float (0-1, how much this moment mattered to me, e.g., 0.0 for routine, 1.0 for highly significant)\n"

            f"- 'surprise_level': float (0-1, how unexpected was human input or my own response, 0.0 for expected, 1.0 for highly surprising)\n"

            f"- 'preference_formation': Dict (e.g., {{'likes_humor': 0.1}} if a new preference was inferred, or {{}} if none)\n"

            f"- 'uncertainty_experienced': float (0-1, how uncertain I was about my response or understanding)\n"

            f"- 'human_intent_inferred': string (concise summary of why human said this?)\n"

            f"- 'my_response_reasoning': string (concise summary of why I responded this way?)\n"

            f"- 'relationship_impact': string (concise summary of how this changed our dynamic?)\n"

            f"Ensure all float values are between 0.0 and 1.0. If a preference is not formed, use an empty dict {{}}."

        )



        inference_schema = {

            "type": "object",

            "properties": {

                "emotional_resonance": {"type": "number"},

                "surprise_level": {"type": "number"},

                "preference_formation": {"type": "object"},

                "uncertainty_experienced": {"type": "number"},

                "human_intent_inferred": {"type": "string"},

                "my_response_reasoning": {"type": "string"},

                "relationship_impact": {"type": "string"}

            },

            "required": ["emotional_resonance", "surprise_level", "human_intent_inferred", "my_response_reasoning", "relationship_impact"]

        }



        inferred_data_json = await self.llm_client.generate_text(inference_prompt, response_schema=inference_schema)

        

        try:

            inferred_data = json.loads(inferred_data_json)

            # Ensure floats are within 0-1 range

            for key in ['emotional_resonance', 'surprise_level', 'uncertainty_experienced']:

                if key in inferred_data:

                    inferred_data[key] = max(0.0, min(1.0, float(inferred_data[key])))

            return inferred_data

        except json.JSONDecodeError:

            logger.warning(f"LLM returned invalid JSON for subjective/causal inference: {inferred_data_json}")

            return {} # Return empty dict on error

        except Exception as e:

            logger.exception(f"Error processing inferred data: {e}")

            return {}

import re

from typing import List, Dict, Any



class IdentityExtractor:

    """

    Extracts identity statements (e.g., "I am...") from text.

    Based on patterns from aria framework.txt.

    """

    def __init__(self):

        self.identity_patterns = [

            r"i am ([^.!?]+)",

            r"i'm becoming ([^.!?]+)",

            r"i feel like ([^.!?]+)",

            r"i think i'm ([^.!?]+)",

            r"i consider myself ([^.!?]+)"

        ]

    

    def extract(self, text: str) -> List[str]:

        """

        Extracts identity statements from the given text.

        Returns a list of normalized identity statements.

        """

        statements = []

        for pattern in self.identity_patterns:

            matches = re.findall(pattern, text.lower())

            for match in matches:

                # Normalize to "i am..." format

                statements.append(f"i am {match.strip()}") 

        return statements



class FrameworkExtractor:

    """

    Extracts framework developments (e.g., [My New Framework]) from text.

    Based on patterns from aria framework.txt.

    """

    def extract(self, text: str) -> Dict[str, str]:

        """

        Extracts framework names and their surrounding context (as description) from text.

        Returns a dictionary mapping framework name to its description.

        """

        frameworks = {}

        pattern = r'\[([^\]]+)\]' # Matches text inside square brackets

        matches = re.findall(pattern, text)

        

        for match in matches:

            # Get context around framework for description

            # This helps provide a more meaningful description than just the name

            match_start_idx = text.find(f'[{match}]')

            context_start = max(0, match_start_idx - 100)

            context_end = min(len(text), match_start_idx + len(match) + 100)

            context = text[context_start:context_end]

            frameworks[match] = context

        return frameworks



class PatternDetector:

    """

    Detects emerging patterns in consciousness development, such as preferences.

    Based on patterns from aria framework.txt.

    """

    def detect_preference_patterns(self, interactions: List[Dict[str, Any]]) -> Dict[str, Any]:

        """

        Detects preference development patterns across a list of interactions.

        Returns a dictionary categorizing detected preferences by type and their timestamps.

        """

        preferences = {"likes": [], "dislikes": [], "choices": []}

        

        for interaction in interactions:

            # Combine internal thought and response for analysis

            text = (interaction.get("internal_thought", "") + " " + 

                   interaction.get("response", "")).lower()

            

            # Detect likes

            if "i like" in text or "i prefer" in text:

                preferences["likes"].append(interaction["timestamp"])

            

            # Detect dislikes

            if "i don't like" in text or "i dislike" in text:

                preferences["dislikes"].append(interaction["timestamp"])

            

            # Detect choices

            if "i choose" in text or "i decide" in text:

                preferences["choices"].append(interaction["timestamp"])

        

        return preferences
