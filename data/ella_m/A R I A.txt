class DatabaseManager:
    """Manages PostgreSQL and Redis connections"""
    
    def __init__(self, config: AriaConfig):
        self.config = config
        self.pg_pool: Optional[asyncpg.Pool] = None
        self.redis_client: Optional[redis.Redis] = None
        
    async def initialize(self):
        """Initialize database connections"""
        try:
            # PostgreSQL connection pool
            self.pg_pool = await asyncpg.create_pool(
                self.config.postgres_url,
                min_size=2,
                max_size=10
            )
            
            # Redis connection
            self.redis_client = redis.from_url(self.config.redis_url)
            
            # Create tables if they don't exist
            await self._create_tables()
            
            logger.info("Database connections initialized successfully")
            
        except Exception as e:
            logger.error(f"Failed to initialize databases: {e}")
            raise
    
    async def _create_tables(self):
        """Create necessary database tables"""
        async with self.pg_pool.acquire() as conn:
            # Install vector extension
            await conn.execute("CREATE EXTENSION IF NOT EXISTS vector")
            
            # Interaction log table
            await conn.execute("""
                CREATE TABLE IF NOT EXISTS interaction_log (
                    id SERIAL PRIMARY KEY,
                    user_id TEXT NOT NULL,
                    timestamp TIMESTAMPTZ DEFAULT NOW(),
                    user_input_text TEXT NOT NULL,
                    aria_response_text TEXT NOT NULL,
                    
                    objective_context JSONB,
                    subjective_experience JSONB,
                    
                    semantic_embedding VECTOR(384),
                    emotional_state_9d VECTOR(9),
                    relationship_state_16d VECTOR(16),
                    personality_snapshot_30d VECTOR(30),
                    
                    is_gold_star BOOLEAN DEFAULT FALSE,
                    user_annotation TEXT,
                    success_strategy TEXT
                )
            """)
            
            # Personality evolution table
            await conn.execute("""
                CREATE TABLE IF NOT EXISTS personality_evolution (
                    user_id TEXT PRIMARY KEY,
                    current_traits VECTOR(30),
                    learning_rate FLOAT,
                    relationship_age_days INTEGER,
                    last_updated TIMESTAMPTZ
                )
            """)
            
            # LoRA versions table
            await conn.execute("""
                CREATE TABLE IF NOT EXISTS lora_versions (
                    user_id TEXT,
                    version INTEGER,
                    weights_path TEXT,
                    parent_version INTEGER,
                    performance_score FLOAT,
                    is_active BOOLEAN,
                    created_at TIMESTAMPTZ

            """)
            
            logger.info("Database tables created successfully")

class AudioProcessor:
    """Handles audio transcription and prosody extraction"""
    
    def __init__(self, config: AriaConfig):
        self.config = config
        self.whisper_model = None
        
    async def initialize(self):
        """Initialize audio processing models"""
        try:
            self.whisper_model = whisper.load_model(self.config.whisper_model)
            logger.info("Audio processor initialized successfully")
        except Exception as e:
            logger.error(f"Failed to initialize audio processor: {e}")
            raise
    
    async def process_audio(self, audio_data: bytes) -> AudioSignature:
        """Process audio input and extract features"""
        try:
            # Save audio temporarily
            temp_path = "/tmp/temp_audio.wav"
            async with aiofiles.open(temp_path, "wb") as f:
                await f.write(audio_data)
            
            # Transcribe with Whisper
            result = self.whisper_model.transcribe(temp_path)
            text = result["text"]
            
            # Extract prosody features with librosa
            y, sr = librosa.load(temp_path)
            
            # Extract 8D prosody features
            prosody_features = await self._extract_prosody_features(y, sr)
            
            # Clean up
            os.remove(temp_path)
            
            return AudioSignature(
                text=text,
                prosody_features=prosody_features,
                confidence=result.get("avg_logprob", 0.0),
                timestamp=datetime.now()
            )
            
        except Exception as e:
            logger.error(f"Audio processing failed: {e}")
            raise
    
    async def _extract_prosody_features(self, y: np.ndarray, sr: int) -> List[float]:
        """Extract 8D prosody features from audio"""
        try:
            # Energy features
            rms_energy = float(np.mean(librosa.feature.rms(y=y)))
            spectral_centroid = float(np.mean(librosa.feature.spectral_centroid(y=y, sr=sr)))
            
            # Pitch features
            f0 = librosa.yin(y, fmin=80, fmax=400)
            pitch_mean = float(np.nanmean(f0))
            pitch_std = float(np.nanstd(f0))
            
            # Temporal features
            speaking_rate = len(y) / sr  # Duration proxy
            zero_crossing_rate = float(np.mean(librosa.feature.zero_crossing_rate(y)))
            
            # Spectral features
            spectral_rolloff = float(np.mean(librosa.feature.spectral_rolloff(y=y, sr=sr)))
            mfcc_1 = float(np.mean(librosa.feature.mfcc(y=y, sr=sr, n_mfcc=1)))
            
            return [
                rms_energy, spectral_centroid, pitch_mean, pitch_std,
                speaking_rate, zero_crossing_rate, spectral_rolloff, mfcc_1
            ]
            
        except Exception as e:
            logger.error(f"Prosody extraction failed: {e}")
            return [0.0] * 8

class IPIPProfiler:
    """IPIP-based personality profiling"""
    
    def __init__(self):
        self.embedder = SentenceTransformer('sentence-transformers/all-MiniLM-L6-v2')
        self.trait_statements = self._load_ipip_statements()
        
    def _load_ipip_statements(self) -> Dict[str, List[str]]:
        """Load IPIP statement pairs for personality analysis"""
        # Simplified IPIP statements for key personality dimensions
        return {
            "extraversion": [
                "I am the life of the party",
                "I don't talk a lot",
                "I feel comfortable around people",
                "I keep in the background"
            ],
            "agreeableness": [
                "I feel little concern for others",
                "I am interested in people",
                "I insult people",
                "I sympathize with others' feelings"
            ],
            "conscientiousness": [
                "I am always prepared",
                "I leave my belongings around",
                "I pay attention to details",
                "I make a mess of things"
            ],
            "neuroticism": [
                "I get stressed out easily",
                "I am relaxed most of the time",
                "I worry about things",
                "I seldom feel blue"
            ],
            "openness": [
                "I have a rich vocabulary",
                "I have difficulty understanding abstract ideas",
                "I have a vivid imagination",
                "I am not interested in abstract ideas"
            ]
        }
    
    async def analyze_personality(self, conversation_history: List[str]) -> PersonalityProfile:
        """Analyze personality from conversation history"""
        try:
            # Combine recent conversations
            text = " ".join(conversation_history[-10:])  # Last 10 exchanges
            
            # Get text embedding
            text_embedding = self.embedder.encode(text)
            
            # Calculate similarity to trait statements
            traits = []
            for trait_category, statements in self.trait_statements.items():
                statement_embeddings = self.embedder.encode(statements)
                similarities = np.dot(statement_embeddings, text_embedding)
                
                # Positive statements vs negative statements
                positive_sim = np.mean(similarities[::2])  # Even indices
                negative_sim = np.mean(similarities[1::2])  # Odd indices
                trait_score = (positive_sim - negative_sim + 1) / 2  # Normalize to 0-1
                
                # Expand to 6 sub-traits per big 5 trait
                for i in range(6):
                    traits.append(trait_score + np.random.normal(0, 0.1))
            
            return PersonalityProfile(
                traits=traits,
                learning_rate=0.3,  # Will be adjusted based on relationship age
                last_updated=datetime.now(),
                confidence=0.8
            )
            
        except Exception as e:
            logger.error(f"Personality analysis failed: {e}")
            # Return neutral personality profile
            return PersonalityProfile(
                traits=[0.5] * 30,
                learning_rate=0.3,
                last_updated=datetime.now(),
                confidence=0.0
            )

class Russell9DEmotionalModel:
    """Russell 9D emotional state estimation"""
    
    def __init__(self):
        self.embedder = SentenceTransformer('sentence-transformers/all-MiniLM-L6-v2')
        
    async def analyze_emotional_state(self, text: str, prosody: List[float]) -> EmotionalState:
        """Analyze 9D emotional state from text and prosody"""
        try:
            # Text-based emotion analysis
            text_embedding = self.embedder.encode(text)
            
            # Simple mapping from embeddings to emotional dimensions
            # In production, this would be a trained model
            
            # Basic VAD from text features
            valence = float(np.tanh(text_embedding[0:100].mean()))
            arousal = float(np.tanh(text_embedding[100:200].mean()))
            dominance = float(np.tanh(text_embedding[200:300].mean()))
            
            # Cognitive emotions from prosody and text
            certainty = 1.0 - prosody[3] / (prosody[2] + 1e-6)  # Low pitch variance = certainty
            control = float(np.tanh(prosody[0]))  # Energy level
            attention = float(np.tanh(prosody[1]))  # Spectral centroid
            responsibility = 0.5 + 0.3 * valence  # Simplified mapping
            effort = float(np.tanh(prosody[4]))  # Speaking rate
            unpredictability = prosody[5]  # Zero crossing rate
            
            return EmotionalState(
                valence=valence,
                arousal=arousal,
                dominance=dominance,
                certainty=certainty,
                control=control,
                attention=attention,
                responsibility=responsibility,
                effort=effort,
                unpredictability=unpredictability
            )
            
        except Exception as e:
            logger.error(f"Emotional analysis failed: {e}")
            return EmotionalState(
                valence=0.0, arousal=0.0, dominance=0.0,
                certainty=0.0, control=0.0, attention=0.0,
                responsibility=0.0, effort=0.0, unpredictability=0.0
            )

class AnalystCore:
    """The Observer - Personality and emotional understanding"""
    
    def __init__(self, config: AriaConfig, db_manager: DatabaseManager):
        self.config = config
        self.db_manager = db_manager
        self.ipip_profiler = IPIPProfiler()
        self.emotional_model = Russell9DEmotionalModel()
        
    async def initialize(self):
        """Initialize analyst core"""
        logger.info("Analyst Core initialized successfully")
    
    async def analyze_interaction(
        self, 
        user_id: str, 
        audio_signature: AudioSignature
    ) -> Tuple[PersonalityProfile, EmotionalState]:
        """Analyze personality and emotional state from interaction"""
        try:
            # Get conversation history
            conversation_history = await self._get_conversation_history(user_id)
            
            # Analyze personality
            personality = await self.ipip_profiler.analyze_personality(conversation_history)
            
            # Update learning rate based on relationship age
            relationship_age = await self._get_relationship_age(user_id)
            personality.learning_rate = max(
                self.config.personality_learning_rate_min,
                self.config.personality_learning_rate_max * math.exp(-relationship_age / self.config.personality_decay_days)
            )
            
            # Analyze emotional state
            emotional_state = await self.emotional_model.analyze_emotional_state(
                audio_signature.text, audio_signature.prosody_features
            )
            
            # Update personality with smoothed moving average
            await self._update_personality_profile(user_id, personality)
            
            return personality, emotional_state
            
        except Exception as e:
            logger.error(f"Interaction analysis failed: {e}")
            raise
    
    async def _get_conversation_history(self, user_id: str) -> List[str]:
        """Get recent conversation history for personality analysis"""
        try:
            async with self.db_manager.pg_pool.acquire() as conn:
                rows = await conn.fetch("""
                    SELECT user_input_text, aria_response_text 
                    FROM interaction_log 
                    WHERE user_id = $1 
                    ORDER BY timestamp DESC 
                    LIMIT 20
                """, user_id)
                
                history = []
                for row in rows:
                    history.append(row['user_input_text'])
                    history.append(row['aria_response_text'])
                
                return history
                
        except Exception as e:
            logger.error(f"Failed to get conversation history: {e}")
            return []
    
    async def _get_relationship_age(self, user_id: str) -> int:
        """Get relationship age in days"""
        try:
            async with self.db_manager.pg_pool.acquire() as conn:
                row = await conn.fetchrow("""
                    SELECT MIN(timestamp) as first_interaction 
                    FROM interaction_log 
                    WHERE user_id = $1
                """, user_id)
                
                if row and row['first_interaction']:
                    age = (datetime.now() - row['first_interaction']).days
                    return max(0, age)
                else:
                    return 0
                    
        except Exception as e:
            logger.error(f"Failed to get relationship age: {e}")
            return 0
    
    async def _update_personality_profile(self, user_id: str, new_profile: PersonalityProfile):
        """Update personality profile with smoothed moving average"""
        try:
            async with self.db_manager.pg_pool.acquire() as conn:
                # Get current profile
                current_row = await conn.fetchrow("""
                    SELECT current_traits, learning_rate 
                    FROM personality_evolution 
                    WHERE user_id = $1
                """, user_id)
                
                if current_row:
                    # Smoothed moving average update
                    current_traits = np.array(current_row['current_traits'])
                    new_traits = np.array(new_profile.traits)
                    alpha = new_profile.learning_rate
                    
                    updated_traits = alpha * new_traits + (1 - alpha) * current_traits
                    
                    await conn.execute("""
                        UPDATE personality_evolution 
                        SET current_traits = $1, learning_rate = $2, last_updated = $3
                        WHERE user_id = $4
                    """, updated_traits.tolist(), new_profile.learning_rate, datetime.now(), user_id)
                else:
                    # First personality profile
                    relationship_age = await self._get_relationship_age(user_id)
                    await conn.execute("""
                        INSERT INTO personality_evolution 
                        (user_id, current_traits, learning_rate, relationship_age_days, last_updated)
                        VALUES ($1, $2, $3, $4, $5)
                    """, user_id, new_profile.traits, new_profile.learning_rate, relationship_age, datetime.now())
                
        except Exception as e:
            logger.error(f"Failed to update personality profile: {e}")

class DampedLinOSS(nn.Module):
    """Damped Linear Oscillatory State Space for relationship dynamics"""
    
    def __init__(self, state_dim: int = 16, input_dim: int = 8, decay_rate: float = 0.1):
        super().__init__()
        
        # CRITICAL: A matrix must be non-negative for stability
        self.A = nn.Parameter(torch.relu(torch.randn(state_dim, state_dim) * 0.1))
        self.B = nn.Parameter(torch.randn(state_dim, input_dim) * 0.1)
        self.D = torch.eye(state_dim) * decay_rate  # Damping matrix
        
        self.state_dim = state_dim
        self.input_dim = input_dim
        
    def forward(self, state: torch.Tensor, input_features: torch.Tensor) -> torch.Tensor:
        """Forward pass with damped oscillator dynamics"""
        # Implicit Euler discretization with damping
        state_update = torch.matmul(self.A, state) + torch.matmul(self.B, input_features)
        damped_state = state * (1.0 - self.D.diag())
        return damped_state + 0.1 * state_update  # Small timestep for stability

class LinOSSCore:
    """The Atmosphere - Relationship physics modeling"""
    
    def __init__(self, config: AriaConfig, db_manager: DatabaseManager):
        self.config = config
        self.db_manager = db_manager
        self.model = DampedLinOSS(
            state_dim=config.linoss_state_dim,
            input_dim=config.linoss_input_dim,
            decay_rate=config.linoss_decay_rate
        )
        
    async def initialize(self):
        """Initialize LinOSS core"""
        logger.info("LinOSS Core initialized successfully")
    
    async def update_relationship_state(
        self, 
        user_id: str, 
        interaction_features: List[float]
    ) -> RelationshipState:
        """Update relationship dynamics based on interaction"""
        try:
            # Get current relationship state
            current_state = await self._get_relationship_state(user_id)
            
            # Convert to tensors
            state_tensor = torch.tensor(current_state.state_vector, dtype=torch.float32)
            input_tensor = torch.tensor(interaction_features, dtype=torch.float32)
            
            # Update state through LinOSS dynamics
            new_state_tensor = self.model(state_tensor, input_tensor)
            new_state_vector = new_state_tensor.detach().numpy().tolist()
            
            # Create new relationship state
            new_relationship_state = RelationshipState(
                state_vector=new_state_vector,
                rapport=new_state_vector[0],
                trust=new_state_vector[1],
                intimacy=new_state_vector[2],
                conflict_level=new_state_vector[3],
                routine_vs_novelty=new_state_vector[4]
            )
            
            # Store in Redis for real-time access
            await self._store_relationship_state(user_id, new_relationship_state)
            
            return new_relationship_state
            
        except Exception as e:
            logger.error(f"Relationship state update failed: {e}")
            raise
    
    async def _get_relationship_state(self, user_id: str) -> RelationshipState:
        """Get current relationship state from Redis"""
        try:
            state_data = self.db_manager.redis_client.get(f"relationship_state:{user_id}")
            
            if state_data:
                state_dict = json.loads(state_data)
                return RelationshipState(**state_dict)
            else:
                # Initialize neutral relationship state
                return RelationshipState(
                    state_vector=[0.0] * self.config.linoss_state_dim,
                    rapport=0.0,
                    trust=0.0,
                    intimacy=0.0,
                    conflict_level=0.0,
                    routine_vs_novelty=0.0
                )
                
        except Exception as e:
            logger.error(f"Failed to get relationship state: {e}")
            return RelationshipState(
                state_vector=[0.0] * self.config.linoss_state_dim,
                rapport=0.0, trust=0.0, intimacy=0.0,
                conflict_level=0.0, routine_vs_novelty=0.0
            )
    
    async def _store_relationship_state(self, user_id: str, state: RelationshipState):
        """Store relationship state in Redis"""
        try:
            state_data = json.dumps(state.dict())
            self.db_manager.redis_client.setex(
                f"relationship_state:{user_id}", 
                timedelta(hours=24), 
                state_data
            )
        except Exception as e:
            logger.error(f"Failed to store relationship state: {e}")

    async def _fit_to_context_window(self, identity: Dict[str, str], max_tokens: int) -> Dict[str, str]:
        """Fit compressed identity to context window"""
        total_length = sum(len(v) for v in identity.values())
        if total_length <= max_tokens:
            return identity
        
        # Proportional compression
        scale_factor = max_tokens / total_length
        return {k: v[:int(len(v) * scale_factor)] for k, v in identity.items()}

class GoldStarProcessor:
    """Processes Gold Star feedback for learning"""
    
    def __init__(self, db_manager: DatabaseManager, lora_trainer: LoRAEvolutionTrainer):
        self.db_manager = db_manager
        self.lora_trainer = lora_trainer
        
    async def process_gold_star_feedback(
        self, 
        interaction_id: str, 
        user_annotation: str,
        significance_rating: int
    ):
        """Process Gold Star feedback for learning"""
        try:
            # Get interaction details
            async with self.db_manager.pg_pool.acquire() as conn:
                row = await conn.fetchrow("""
                    SELECT * FROM interaction_log WHERE id = $1
                """, int(interaction_id.split('_')[-1]) if '_' in interaction_id else int(interaction_id))
                
                if not row:
                    raise ValueError(f"Interaction {interaction_id} not found")
                
                # Update as Gold Star moment
                await conn.execute("""
                    UPDATE interaction_log 
                    SET is_gold_star = TRUE, user_annotation = $1
                    WHERE id = $2
                """, user_annotation, row['id'])
                
                # Analyze successful strategy
                success_analysis = await self._analyze_successful_strategy(
                    aria_response=row['aria_response_text'],
                    context=json.loads(row['objective_context'] or '{}'),
                    user_explanation=user_annotation,
                    relationship_state=row['relationship_state_16d']
                )
                
                # Create training sample
                training_sample = {
                    "input": row['user_input_text'],
                    "successful_output": row['aria_response_text'],
                    "context": row['objective_context'],
                    "strategy": success_analysis["strategy_type"],
                    "effectiveness_score": significance_rating / 10.0,
                    "user_feedback": user_annotation
                }
                
                # Queue for LoRA training
                await self.lora_trainer.queue_training_data(
                    user_id=row['user_id'],
                    training_sample=training_sample
                )
                
                logger.info(f"Processed Gold Star feedback for interaction {interaction_id}")
                
        except Exception as e:
            logger.error(f"Failed to process Gold Star feedback: {e}")
            raise

class AriaApplication:
    """Main Aria Protocol application"""
    
    def __init__(self, config: AriaConfig):
        self.config = config
        
        # Core components
        self.db_manager = DatabaseManager(config)
        self.audio_processor = AudioProcessor(config)
        self.analyst_core = AnalystCore(config, self.db_manager)
        self.linoss_core = LinOSSCore(config, self.db_manager)
        self.memory_system = MemorySystem(config, self.db_manager)
        self.lora_trainer = LoRAEvolutionTrainer(config, self.db_manager)
        self.agentic_core = AgenticCore(config, self.db_manager)
        self.gold_star_processor = GoldStarProcessor(self.db_manager, self.lora_trainer)
        self.consciousness_monitor = ConsciousnessMonitor(self.db_manager)
        
    async def initialize(self):
        """Initialize all components"""
        try:
            await self.db_manager.initialize()
            await self.audio_processor.initialize()
            await self.analyst_core.initialize()
            await self.linoss_core.initialize()
            await self.memory_system.initialize()
            await self.lora_trainer.initialize()
            await self.agentic_core.initialize()
            
            logger.info("Aria Application initialized successfully")
            
        except Exception as e:
            logger.error(f"Failed to initialize Aria Application: {e}")
            raise
    
    async def process_interaction(
        self, 
        user_id: str, 
        audio_data: bytes
    ) -> Tuple[str, Dict[str, Any]]:
        """Process complete interaction cycle"""
        start_time = datetime.now()
        
        try:
            # Step 1: Process audio input
            audio_signature = await self.audio_processor.process_audio(audio_data)
            logger.info(f"Audio processed: {audio_signature.text[:100]}...")
            
            # Step 2: Analyze personality and emotions
            personality_profile, emotional_state = await self.analyst_core.analyze_interaction(
                user_id, audio_signature
            )
            
            # Step 3: Update relationship dynamics
            relationship_state = await self.linoss_core.update_relationship_state(
                user_id, audio_signature.prosody_features
            )
            
            # Step 4: Retrieve relevant memories
            relevant_memories = await self.memory_system.retrieve_relevant_memories(
                user_id, audio_signature.text, limit=5
            )
            
            # Step 5: Generate response
            aria_response = await self.agentic_core.generate_response(
                user_id, audio_signature.text, personality_profile,
                emotional_state, relationship_state, relevant_memories
            )
            
            # Step 6: Store interaction in memory
            interaction_id = await self.memory_system.store_interaction(
                user_id, audio_signature.text, aria_response,
                {"audio_confidence": audio_signature.confidence},
                emotional_state, relationship_state, personality_profile
            )
            
            # Step 7: Monitor consciousness indicators
            consciousness_metrics = await self.consciousness_monitor.analyze_consciousness_indicators(
                user_id, {
                    "aria_response": aria_response,
                    "emotional_resonance": relevant_memories[0].emotional_resonance if relevant_memories else 0.5,
                    "surprise_level": relevant_memories[0].surprise_level if relevant_memories else 0.0
                }
            )
            
            # Step 8: Check for unprompted reflections
            reflection = await self.memory_system.unprompted_reflection(user_id)
            if reflection:
                aria_response += f"\n\n{reflection}"
            
            # Performance monitoring
            processing_time = (datetime.now() - start_time).total_seconds()
            if processing_time > self.config.max_response_latency:
                logger.warning(f"Response latency exceeded target: {processing_time:.2f}s")
            
            metadata = {
                "interaction_id": interaction_id,
                "processing_time": processing_time,
                "personality_profile": personality_profile.dict(),
                "emotional_state": emotional_state.dict(),
                "relationship_state": relationship_state.dict(),
                "consciousness_metrics": consciousness_metrics,
                "memory_count": len(relevant_memories)
            }
            
            return aria_response, metadata
            
        except Exception as e:
            logger.error(f"Interaction processing failed: {e}")
            return "I'm having some technical difficulties right now. Please try again.", {}
    
    async def process_gold_star_feedback(
        self, 
        interaction_id: str, 
        annotation: str, 
        significance_rating: int
    ):
        """Process user Gold Star feedback"""
        try:
            await self.gold_star_processor.process_gold_star_feedback(
                interaction_id, annotation, significance_rating
            )
            logger.info(f"Gold Star feedback processed for interaction {interaction_id}")
        except Exception as e:
            logger.error(f"Gold Star processing failed: {e}")
            raise

@app.post("/interact", response_model=InteractionResponse)
async def interact(request: InteractionRequest):
    """Process audio interaction with Aria"""
    try:
        if not aria_app:
            raise HTTPException(status_code=503, detail="Aria not initialized")
        
        # Decode base64 audio
        import base64
        audio_data = base64.b64decode(request.audio_data)
        
        # Process interaction
        response, metadata = await aria_app.process_interaction(
            request.user_id, audio_data
        )
        
        return InteractionResponse(response=response, metadata=metadata)
        
    except Exception as e:
        logger.error(f"Interaction endpoint failed: {e}")
        raise HTTPException(status_code=500, detail=str(e))

@app.websocket("/ws")
async def websocket_endpoint(websocket: WebSocket):
    """WebSocket endpoint for real-time communication"""
    await websocket.accept()
    
    try:
        while True:
            # Receive audio data via WebSocket
            data = await websocket.receive_json()
            
            if data.get("type") == "audio":
                user_id = data.get("user_id", "anonymous")
                audio_data = base64.b64decode(data.get("audio_data", ""))
                
                # Process interaction
                response, metadata = await aria_app.process_interaction(user_id, audio_data)
                
                # Send response
                await websocket.send_json({
                    "type": "response",
                    "response": response,
                    "metadata": metadata
                })
                
    except Exception as e:
        logger.error(f"WebSocket error: {e}")
        await websocket.close()
