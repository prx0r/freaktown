pythonimport torch
import torch.nn as nn
import numpy as np
from typing import Dict, List, Tuple
import redis
import json

class DampedLinOSS(nn.Module):
    """Damped Linear Oscillator State Space model for relationship dynamics"""
    
    def __init__(self, state_dim: int = 16, input_dim: int = 8, decay_rate: float = 0.1):
        super().__init__()
        self.state_dim = state_dim
        self.input_dim = input_dim
        self.decay_rate = decay_rate
        
        # Learnable parameters
        self.A = nn.Parameter(torch.randn(state_dim, state_dim) * 0.1)  # State transition
        self.B = nn.Parameter(torch.randn(state_dim, input_dim) * 0.1)  # Input mapping
        self.C = nn.Parameter(torch.randn(state_dim, state_dim) * 0.1)  # Output mapping
        
        # Damping matrix (diagonal for stability)
        self.register_buffer('D', torch.eye(state_dim) * decay_rate)
        
    def forward(self, state: torch.Tensor, input_event: torch.Tensor) -> torch.Tensor:
        """
        Update state based on new interaction event
        
        Args:
            state: Current relationship state [state_dim]
            input_event: New interaction features [input_dim]
            
        Returns:
            Updated state [state_dim]
        """
        # Oscillator dynamics: dx/dt = Ax + Bu - Dx (damped)
        state_update = torch.matmul(self.A, state) + torch.matmul(self.B, input_event)
        damped_state = state - self.D * state  # Apply damping
        
        new_state = damped_state + 0.1 * state_update  # Euler integration with small step
        
        # Apply nonlinearity to prevent explosion
        new_state = torch.tanh(new_state)
        
        return new_state
    
    def get_context_vector(self, state: torch.Tensor) -> torch.Tensor:
        """Convert internal state to context vector for LLM"""
        return torch.matmul(self.C, state)

class RelationalCore:
    """Manages LinOSS state and Redis persistence"""
    
    def __init__(self):
        self.redis_client = redis.Redis.from_url(config.REDIS_URL)
        self.linoss_model = DampedLinOSS(
            state_dim=config.LINOSS_STATE_DIM,
            input_dim=8,  # Prosody features
            decay_rate=config.LINOSS_DECAY_RATE
        )
        
    def get_state(self, user_id: str) -> torch.Tensor:
        """Load current relationship state from Redis"""
        key = f"linoss_state:{user_id}"
        state_json = self.redis_client.get(key)
        
        if state_json:
            state_list = json.loads(state_json)
            return torch.tensor(state_list, dtype=torch.float32)
        else:
            # Initialize with small random state
            return torch.randn(config.LINOSS_STATE_DIM) * 0.1
    
    def update_state(self, user_id: str, prosody_features: List[float]) -> torch.Tensor:
        """Update relationship state with new interaction"""
        current_state = self.get_state(user_id)
        input_event = torch.tensor(prosody_features, dtype=torch.float32)
        
        # Update state through LinOSS
        with torch.no_grad():
            new_state = self.linoss_model(current_state, input_event)
        
        # Save to Redis
        key = f"linoss_state:{user_id}"
        self.redis_client.setex(key, 3600, json.dumps(new_state.tolist()))  # 1 hour TTL
        
        return new_state
    
    def get_context_vector(self, user_id: str) -> torch.Tensor:
        """Get context vector for LLM prompt"""
        state = self.get_state(user_id)
        with torch.no_grad():
            return self.linoss_model.get_context_vector(state)

class AudioProcessor:
    """Handles audio transcription and prosody extraction"""
    
    def __init__(self):
        self.whisper_model = whisper.load_model(config.WHISPER_MODEL)
        
    def transcribe_audio(self, audio_bytes: bytes) -> str:
        """Transcribe audio to text using Whisper"""
        # Convert bytes to numpy array
        audio_array = np.frombuffer(audio_bytes, dtype=np.float32)
        
        # Whisper expects 16kHz
        if len(audio_array) > 0:
            result = self.whisper_model.transcribe(audio_array)
            return result["text"].strip()
        return ""
    
    def extract_prosody(self, audio_bytes: bytes) -> List[float]:
        """Extract prosodic features from audio"""
        try:
            # Convert to numpy array
            audio_array = np.frombuffer(audio_bytes, dtype=np.float32)
            
            if len(audio_array) == 0:
                return [0.0] * 8  # Return zeros if no audio
            
            # Ensure we have enough samples
            if len(audio_array) < 1024:
                audio_array = np.pad(audio_array, (0, 1024 - len(audio_array)))
            
            sr = 16000  # Assume 16kHz sample rate
            
            # Extract features
            features = []
            
            # 1. Pitch (F0) - fundamental frequency
            pitches, magnitudes = librosa.piptrack(y=audio_array, sr=sr, threshold=0.1)
            pitch_mean = np.mean(pitches[pitches > 0]) if np.any(pitches > 0) else 0.0
            features.append(float(pitch_mean))
            
            # 2. Energy (RMS)
            rms = librosa.feature.rms(y=audio_array)[0]
            energy_mean = np.mean(rms)
            features.append(float(energy_mean))
            
            # 3. Zero crossing rate (speech activity)
            zcr = librosa.feature.zero_crossing_rate(audio_array)[0]
            zcr_mean = np.mean(zcr)
            features.append(float(zcr_mean))
            
            # 4. Spectral centroid (brightness)
            spectral_centroid = librosa.feature.spectral_centroid(y=audio_array, sr=sr)[0]
            sc_mean = np.mean(spectral_centroid)
            features.append(float(sc_mean))
            
            # 5. Spectral rolloff
            spectral_rolloff = librosa.feature.spectral_rolloff(y=audio_array, sr=sr)[0]
            sr_mean = np.mean(spectral_rolloff)
            features.append(float(sr_mean))
            
            # 6. Tempo
            tempo, _ = librosa.beat.beat_track(y=audio_array, sr=sr)
            features.append(float(tempo))
            
            # 7-8. MFCC first two coefficients
            mfccs = librosa.feature.mfcc(y=audio_array, sr=sr, n_mfcc=2)
            features.extend([float(np.mean(mfccs[0])), float(np.mean(mfccs[1]))])
            
            # Normalize features to [-1, 1] range
            features = [np.tanh(f / 1000.0) for f in features]
            
            return features
            
        except Exception as e:
            print(f"Error extracting prosody: {e}")
            return [0.0] * 8  # Return zeros on error

class SimpleAnalyst:
    """MVP version: prompt-based personality assessment"""
    
    def __init__(self):
        self.big_five_dimensions = [
            "openness", "conscientiousness", "extraversion", 
            "agreeableness", "neuroticism"
        ]
        
    def analyze_personality(self, signature: BehavioralSignature) -> Dict[str, float]:
        """
        Simple personality assessment based on text and prosody
        Returns scores for Big Five traits (0-1 scale)
        """
        transcript = signature.transcript.lower()
        prosody = signature.prosody_features
        
        # Simple heuristics for MVP
        scores = {}
        
        # Openness - creative/curious language
        openness_words = ["creative", "imagine", "wonder", "curious", "new", "different"]
        openness_score = sum(1 for word in openness_words if word in transcript) / 10.0
        openness_score += (prosody[3] + 1) / 4.0  # spectral centroid (brightness)
        scores["openness"] = min(1.0, openness_score)
        
        # Conscientiousness - organized/disciplined language
        conscientiousness_words = ["plan", "organize", "schedule", "goal", "task", "complete"]
        conscientiousness_score = sum(1 for word in conscientiousness_words if word in transcript) / 10.0
        conscientiousness_score += (prosody[5] + 1) / 4.0  # tempo regularity
        scores["conscientiousness"] = min(1.0, conscientiousness_score)
        
        # Extraversion - social/energetic language and prosody
        extraversion_words = ["people", "party", "social", "excited", "energy", "talk"]
        extraversion_score = sum(1 for word in extraversion_words if word in transcript) / 10.0
        extraversion_score += (prosody[1] + 1) / 2.0  # energy/volume
        scores["extraversion"] = min(1.0, extraversion_score)
        
        # Agreeableness - cooperative/trusting language
        agreeableness_words = ["help", "kind", "please", "thank", "agree", "understand"]
        agreeableness_score = sum(1 for word in agreeableness_words if word in transcript) / 10.0
        agreeableness_score += (1 - abs(prosody[0])) / 2.0  # stable pitch
        scores["agreeableness"] = min(1.0, agreeableness_score)
        
        # Neuroticism - anxious/stress language and prosody
        neuroticism_words = ["worry", "stress", "anxious", "nervous", "scared", "difficult"]
        neuroticism_score = sum(1 for word in neuroticism_words if word in transcript) / 10.0
        neuroticism_score += abs(prosody[2]) / 2.0  # zero crossing rate (shakiness)
        scores["neuroticism"] = min(1.0, neuroticism_score)
        
        return scores
    
    def extract_beliefs(self, transcript: str) -> List[str]:
        """Extract stated beliefs from transcript"""
        belief_indicators = [
            "i believe", "i think", "i feel", "i value", 
            "important to me", "i care about", "i love", "i hate"
        ]
        
        beliefs = []
        transcript_lower = transcript.lower()
        
        for indicator in belief_indicators:
            if indicator in transcript_lower:
                # Extract sentence containing the belief
                sentences = transcript.split('.')
                for sentence in sentences:
                    if indicator in sentence.lower():
                        beliefs.append(sentence.strip())
        
        return beliefs
    
    def get_context_package(self, user_id: str, signature: BehavioralSignature) -> Dict:
        """Get relevant context for LLM prompt"""
        # Get recent interactions
        db = next(get_db())
        recent_interactions = db.query(InteractionLog).filter(
            InteractionLog.user_id == user_id
        ).order_by(InteractionLog.timestamp.desc()).limit(5).all()
        
        # Get relevant beliefs
        beliefs = db.query(BeliefGraphNode).filter(
            BeliefGraphNode.user_id == user_id
        ).order_by(BeliefGraphNode.strength.desc()).limit(3).all()
        
        context = {
            "personality_scores": self.analyze_personality(signature),
            "recent_interactions": [
                {"user": int_.user_input, "ai": int_.ai_response} 
                for int_ in recent_interactions
            ],
            "core_beliefs": [belief.belief_text for belief in beliefs],
            "current_prosody": signature.prosody_features
        }
        
        return context

pythonimport torch
from transformers import AutoTokenizer, AutoModelForCausalLM
from peft import PeftModel, LoraConfig, get_peft_model
import os
import json
from typing import Dict, List, Optional

from src.config import config
from src.db.database import get_db, LoRAVersion

class AgenticCore:
    """Manages LLM generation with persistent LoRA adapters"""
    
    def __init__(self):
        self.tokenizer = AutoTokenizer.from_pretrained(config.BASE_MODEL)
        self.base_model = AutoModelForCausalLM.from_pretrained(
            config.BASE_MODEL,
            torch_dtype=torch.float16,
            device_map="auto"
        )
        
        # Add padding token if missing
        if self.tokenizer.pad_token is None:
            self.tokenizer.pad_token = self.tokenizer.eos_token
        
        self.current_user_id = None
        self.current_lora_model = None
        
    def load_user_lora(self, user_id: str) -> bool:
        """Load user's current LoRA adapter"""
        try:
            # Get current LoRA version from database
            db = next(get_db())
            current_version = db.query(LoRAVersion).filter(
                LoRAVersion.user_id == user_id,
                LoRAVersion.is_active == True
            ).first()
            
            if current_version and os.path.exists(current_version.storage_path):
                # Load existing LoRA
                self.current_lora_model = PeftModel.from_pretrained(
                    self.base_model,
                    current_version.storage_path
                )
                print(f"Loaded LoRA {current_version.version} for user {user_id}")
            else:
                # Create new LoRA for user
                self.current_lora_model = self._create_new_lora(user_id)
                print(f"Created new LoRA for user {user_id}")
            
            self.current_user_id = user_id
            return True
            
        except Exception as e:
            print(f"Error loading LoRA for user {user_id}: {e}")
            return False
    
    def _create_new_lora(self, user_id: str) -> PeftModel:
        """Create new LoRA adapter for user"""
        lora_config = LoraConfig(
            r=config.LORA_RANK,
            lora_alpha=config.LORA_ALPHA,
            target_modules=["q_proj", "v_proj"],  # Common attention modules
            lora_dropout=0.1,
            bias="none",
            task_type="CAUSAL_LM"
        )
        
        lora_model = get_peft_model(self.base_model, lora_config)
        
        # Save to database
        lora_path = os.path.join(config.LORA_DIR, f"{user_id}_lora_v1.0")
        os.makedirs(lora_path, exist_ok=True)
        
        lora_model.save_pretrained(lora_path)
        
        # Record in database
        db = next(get_db())
        lora_version = LoRAVersion(
            user_id=user_id,
            version="v1.0",
            storage_path=lora_path,
            is_active=True
        )
        db.add(lora_version)
        db.commit()
        
        return lora_model
    
    def generate_response(self, 
                         user_input: str, 
                         context: Dict, 
                         relationship_context: torch.Tensor) -> str:
        """Generate response using current LoRA"""
        
        if self.current_lora_model is None:
            return "Error: No LoRA loaded"
        
        # Build comprehensive prompt
        prompt = self._build_prompt(user_input, context, relationship_context)
        
        # Tokenize
        inputs = self.tokenizer(
            prompt, 
            return_tensors="pt", 
            padding=True, 
            truncation=True, 
            max_length=512
        ).to(self.current_lora_model.device)
        
        # Generate
        with torch.no_grad():
            outputs = self.current_lora_model.generate(
                **inputs,
                max_new_tokens=150,
                temperature=0.7,
                do_sample=True,
                pad_token_id=self.tokenizer.eos_token_id
            )
        
        # Decode response
        response = self.tokenizer.decode(outputs[0], skip_special_tokens=True)
        
        # Extract only the new part (after the prompt)
        response = response[len(prompt):].strip()
        
        return response
    
    def _build_prompt(self, 
                     user_input: str, 
                     context: Dict, 
                     relationship_context: torch.Tensor) -> str:
        """Build comprehensive prompt for LLM"""
        
        # Convert relationship context to interpretable values
        rel_ctx = relationship_context.cpu().numpy()
        relationship_summary = self._interpret_relationship_context(rel_ctx)
        
        prompt = f"""You are an AI companion having a conversation with your user. 

User's Personality Profile:
- Openness: {context['personality_scores']['openness']:.2f}
- Conscientiousness: {context['personality_scores']['conscientiousness']:.2f}
- Extraversion: {context['personality_scores']['extraversion']:.2f}
- Agreeableness: {context['personality_scores']['agreeableness']:.2f}
- Neuroticism: {context['personality_scores']['neuroticism']:.2f}

Core Beliefs:
{chr(10).join(f"- {belief}" for belief in context['core_beliefs'])}

Relationship Context: {relationship_summary}

Recent Conversation:
{chr(10).join(f"User: {turn['user']}" + chr(10) + f"AI: {turn['ai']}" for turn in context['recent_interactions'][-2:])}

User: {user_input}
AI: """
        
        return prompt
    
    def _interpret_relationship_context(self, rel_ctx: np.ndarray) -> str:
        """Convert relationship context vector to human-readable description"""
        # Simple interpretation for MVP
        energy_level = float(np.mean(rel_ctx[:4]))
        emotional_valence = float(np.mean(rel_ctx[4:8]))
        intimacy_level = float(np.mean(rel_ctx[8:12]))
        conversation_flow = float(np.mean(rel_ctx[12:16]))
        
        descriptions = []
        
        if energy_level > 0.3:
            descriptions.append("high energy")
        elif energy_level < -0.3:
            descriptions.append("low energy")
        
        if emotional_valence > 0.3:
            descriptions.append("positive mood")
        elif emotional_valence < -0.3:
            descriptions.append("negative mood")
        
        if intimacy_level > 0.3:

class InteractionDataset(Dataset):
    """Dataset for LoRA fine-tuning"""
    
    def __init__(self, interactions: List[InteractionLog], tokenizer):
        self.interactions = interactions
        self.tokenizer = tokenizer
        
    def __len__(self):
        return len(self.interactions)
    
    def __getitem__(self, idx):
        interaction = self.interactions[idx]
        
        # Create training example: user input -> AI response
        text = f"User: {interaction.user_input}\nAI: {interaction.ai_response}"
        
        # Tokenize
        encoding = self.tokenizer(
            text,
            truncation=True,
            padding='max_length',
            max_length=512,
            return_tensors='pt'
        )
        
        return {
            'input_ids': encoding['input_ids'].flatten(),
            'attention_mask': encoding['attention_mask'].flatten(),
            'labels': encoding['input_ids'].flatten()
        }

class OfflineTrainer:
    """Handles offline LoRA fine-tuning"""
    
    def __init__(self):
        self.db = SessionLocal()
        
    def get_training_data(self, user_id: str, days_back: int = 7) -> List[InteractionLog]:
        """Get high-quality interactions for training"""
        cutoff_date = datetime.now() - timedelta(days=days_back)
        
        interactions = self.db.query(InteractionLog).filter(
            InteractionLog.user_id == user_id,
            InteractionLog.timestamp >= cutoff_date,
            InteractionLog.feedback_score >= 4  # Only good interactions
        ).all()
        
        return interactions
    
    def train_user_lora(self, user_id: str) -> bool:
        """Train new LoRA version for user"""
        try:
            # Get training data
            training_interactions = self.get_training_data(user_id)
            
            if len(training_interactions) < 5:
                print(f"Not enough training data for user {user_id}")
                return False
            
            # Load current LoRA
            agentic_core = AgenticCore()
            success = agentic_core.load_user_lora(user_id)
            if not success:
                return False
            
            # Create dataset
            dataset = InteractionDataset(training_interactions, agentic_core.tokenizer)
            
            # Get current version
            current_version = self.db.query(LoRAVersion).filter(
                LoRAVersion.user_id == user_id,
                LoRAVersion.is_active == True
            ).first()
            
            if not current_version:
                return False
            
            # Parse version number
            version_num = float(current_version.version.replace('v', ''))
            new_version = f"v{version_num + 0.1:.1f}"
            
            # Setup training
            training_args = TrainingArguments(
                output_dir=f"./tmp_training_{user_id}",
                per_device_train_batch_size=config.BATCH_SIZE,
                num_train_epochs=3,
                learning_rate=config.LEARNING_RATE,
                logging_steps=10,
                save_strategy="no",  # Don't save intermediate checkpoints
                remove_unused_columns=False,
            )
            
            trainer = Trainer(
                model=agentic_core.current_lora_model,
                args=training_args,
                train_dataset=dataset,
                tokenizer=agentic_core.tokenizer,
            )
            
            # Train
            print(f"Training {new_version} for user {user_id}...")
            trainer.train()
            
            # Save new LoRA version
            new_lora_path = os.path.join(config.LORA_DIR, f"{user_id}_lora_{new_version}")
            os.makedirs(new_lora_path, exist_ok=True)
            agentic_core.current_lora_model.save_pretrained(new_lora_path)
            
            # Deactivate old version
            current_version.is_active = False
            
            # Create new version record
            new_lora_record = LoRAVersion(
                user_id=user_id,
                version=new_version,
                parent_version=current_version.version,
                storage_path=new_lora_path,
                training_interaction_ids=[int_.id for int_ in training_interactions],
                performance_metrics=json.dumps({"training_loss": trainer.state.log_history[-1].get("train_loss", 0.0)}),
                is_active=True
            )
            
            self.db.add(new_lora_record)
            
            # Update user's current version
            user = self.db.query(User).filter(User.user_id == user_id).first()
            user.current_lora_version = new_version
            
            self.db.commit()
            
            # Cleanup
            import shutil
            shutil.rmtree(f"./tmp_training_{user_id}", ignore_errors=True)
            
            print(f"Successfully trained {new_version} for user {user_id}")
            return True
            
        except Exception as e:
            print(f"Error training LoRA for user {user_id}: {e}")
            self.db.rollback()
            return False
    
    def train_all_users(self):
        """Train LoRAs for all active users"""
        users = self.db.query(User).all()
        
        for user in users:
            print(f"Processing user {user.user_id}...")
            self.train_user_lora(user.user_id)
    
    def __del__(self):
        self.db.close()

if __name__ == "__main__":

    trainer.train_all_users()

class SamanthaClient:
    def __init__(self, base_url: str = "http://localhost:8000"):
        self.base_url = base_url
        
    def health_check(self):
        """Test if API is running"""
        response = requests.get(f"{self.base_url}/health")
        return response.json()
    
    def interact(self, user_id: str, audio_file_path: str):
        """Send audio interaction"""
        with open(audio_file_path, 'rb') as f:
            files = {'audio_file': f}
            data = {'user_id': user_id}
            
            response = requests.post(
                f"{self.base_url}/api/v1/interact",
                files=files,
                data=data
            )
            
        return response.json()
    
    def submit_feedback(self, interaction_id: int, score: int):
        """Submit feedback for interaction"""
        response = requests.post(
            f"{self.base_url}/api/v1/feedback",
            json={'interaction_id': interaction_id, 'score': score}
        )
        return response.json()
    
    def get_profile(self, user_id: str):
        """Get user personality profile"""
        response = requests.get(f"{self.base_url}/api/v1/user/{user_id}/profile")
        return response.json()

pythonimport psutil
import time
import redis
import psycopg2
from datetime import datetime

from src.config import config

def check_system_resources():
    """Monitor system resources"""
    cpu_percent = psutil.cpu_percent(interval=1)
    memory = psutil.virtual_memory()
    disk = psutil.disk_usage('/')
    
    print(f"CPU: {cpu_percent}%")
    print(f"Memory: {memory.percent}% ({memory.used / 1024**3:.1f}GB / {memory.total / 1024**3:.1f}GB)")
    print(f"Disk: {disk.percent}% ({disk.used / 1024**3:.1f}GB / {disk.total / 1024**3:.1f}GB)")

def check_redis():
    """Check Redis connection and stats"""
    try:
        r = redis.Redis.from_url(config.REDIS_URL)
        info = r.info()
        print(f"Redis: Connected, {info['connected_clients']} clients, {info['used_memory_human']} memory")
        return True
    except Exception as e:
        print(f"Redis: Error - {e}")
        return False

def check_postgres():
    """Check PostgreSQL connection and stats"""
    try:
        conn = psycopg2.connect(config.DATABASE_URL)
        cursor = conn.cursor()
        
        # Get interaction count
        cursor.execute("SELECT COUNT(*) FROM interaction_log")
        interaction_count = cursor.fetchone()[0]
        
        # Get user count
        cursor.execute("SELECT COUNT(*) FROM users")
        user_count = cursor.fetchone()[0]
        
        print(f"PostgreSQL: Connected, {user_count} users, {interaction_count} interactions")
        
        cursor.close()
        conn.close()
        return True
    except Exception as e:
        print(f"PostgreSQL: Error - {e}")
        return False

def monitor_loop():
    """Main monitoring loop"""
    while True:
        print(f"\n=== System Status - {datetime.now()} ===")
        
        check_system_resources()
        redis_ok = check_redis()
        postgres_ok = check_postgres()
        
        if not redis_ok or not postgres_ok:
            print("⚠️  Some services are down!")
        else:
            print("✅ All services running")
        
        time.sleep(30)  # Check every 30 seconds

if __name__ == "__main__":
    monitor_loop()
