import redis.asyncio as redis
import asyncpg
import chromadb
from src.config import settings
import logging

logger = logging.getLogger(__name__)

class DatabaseManager:
    """Manages connections to PostgreSQL, Redis, and ChromaDB."""
    def __init__(self):
        self.db_pool: asyncpg.Pool = None
        self.redis_pool: redis.Redis = None
        self.chroma_client: chromadb.HttpClient = None

    async def connect(self):
        try:
            logger.info("Connecting to PostgreSQL...")
            self.db_pool = await asyncpg.create_pool(dsn=settings.database_url)
            logger.info("Connecting to Redis...")
            self.redis_pool = redis.from_url(settings.redis_url, encoding="utf-8", decode_responses=True)
            await self.redis_pool.ping()
            logger.info("Connecting to ChromaDB...")
            self.chroma_client = chromadb.HttpClient(host=settings.CHROMA_HOST, port=settings.CHROMA_PORT)
            self.chroma_client.heartbeat()
            logger.info("All database connections established.")
            await self._initialize_db()
        except Exception as e:
            logger.critical(f"Failed to connect to databases: {e}", exc_info=True)
            raise

    async def disconnect(self):
        if self.db_pool: await self.db_pool.close()
        if self.redis_pool: await self.redis_pool.close()
        logger.info("Database connections closed.")

    async def _initialize_db(self):
        logger.info("Initializing database schema...")
        async with self.db_pool.acquire() as conn:
            await conn.execute("CREATE EXTENSION IF NOT EXISTS vector;")
            await conn.execute("""
                CREATE TABLE IF NOT EXISTS users (
                    user_id TEXT PRIMARY KEY,
                    created_at TIMESTAMPTZ DEFAULT NOW()
                );
            """)
            await conn.execute("""
                CREATE TABLE IF NOT EXISTS interactions (
                    interaction_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
                    user_id TEXT REFERENCES users(user_id),
                    timestamp TIMESTAMPTZ DEFAULT NOW(),
                    transcript TEXT,
                    aria_response TEXT,
                    arousal REAL,
                    valence REAL
                );
            """)
            await conn.execute("""
                CREATE TABLE IF NOT EXISTS gold_star_data (
                    gold_star_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
                    interaction_id UUID REFERENCES interactions(interaction_id),
                    user_id TEXT REFERENCES users(user_id),
                    annotation TEXT,
                    is_pillar_memory BOOLEAN DEFAULT FALSE,
                    created_at TIMESTAMPTZ DEFAULT NOW(),
                    is_processed BOOLEAN DEFAULT FALSE
                );
            """)
        logger.info("Database schema initialized.")

import torch
import whisper
import numpy as np
import logging
from src.config import settings
import soundfile as sf
import io

logger = logging.getLogger(__name__)

class AudioProcessor:
    """Handles the audio-first pipeline: transcription and prosody analysis."""
    def __init__(self, device="cuda" if torch.cuda.is_available() else "cpu"):
        self.device = device
        try:
            logger.info(f"Loading Whisper model '{settings.WHISPER_MODEL_NAME}' on device '{self.device}'...")
            self.whisper_model = whisper.load_model(settings.WHISPER_MODEL_NAME, device=self.device)
            logger.info("Whisper model loaded successfully.")
        except Exception as e:
            logger.critical(f"Failed to load Whisper model: {e}", exc_info=True)
            raise

    def process_audio(self, audio_bytes: bytes) -> tuple[str, np.ndarray]:
        """Processes raw audio bytes into a transcript and prosody feature vector."""
        try:
            # Use soundfile to read audio bytes, which supports more formats
            audio_np, sample_rate = sf.read(io.BytesIO(audio_bytes))
            if audio_np.ndim > 1:
                audio_np = np.mean(audio_np, axis=1) # Convert to mono

            # Whisper expects float32
            audio_np = audio_np.astype(np.float32)

            # Transcribe audio to text
            result = self.whisper_model.transcribe(audio_np, fp16=(self.device=="cuda"))
            transcript = result['text']

            # Basic prosody features
            energy = np.mean(np.square(audio_np))
            zero_crossing_rate = np.mean(librosa.feature.zero_crossing_rate(y=audio_np))
            prosody_features = np.array([energy, zero_crossing_rate])

            logger.info(f"Transcription successful. Transcript length: {len(transcript)}")
            return transcript, prosody_features
        except Exception as e:
            logger.error(f"Error processing audio: {e}", exc_info=True)
            return "", np.array([0.0, 0.0])

import torch
from transformers import AutoTokenizer, AutoModelForSequenceClassification
import numpy as np
import logging
from src.config import settings

logger = logging.getLogger(__name__)

class AnalystCore:
    """Uses a real, pre-trained model to analyze Arousal and Valence."""
    def __init__(self, device="cuda" if torch.cuda.is_available() else "cpu"):
        self.device = device
        try:
            logger.info(f"Loading emotion model '{settings.EMOTION_MODEL_NAME}' on device '{self.device}'...")
            self.tokenizer = AutoTokenizer.from_pretrained(settings.EMOTION_MODEL_NAME)
            self.model = AutoModelForSequenceClassification.from_pretrained(settings.EMOTION_MODEL_NAME).to(self.device)
            self.model.eval()
            logger.info("Emotion model loaded successfully.")
        except Exception as e:
            logger.critical(f"Failed to load emotion model: {e}", exc_info=True)
            raise
        
        # V: [-1, 1], A: [0, 1]
        self.emotion_to_va = {
            "joy": [0.8, 0.6], "surprise": [0.7, 0.4], "anger": [0.8, -0.7],
            "fear": [0.9, -0.6], "sadness": [0.2, -0.8], "disgust": [0.4, -0.7],
            "neutral": [0.1, 0.0]
        }

    @torch.no_grad()
    def analyze_arousal_valence(self, text: str) -> np.ndarray:
        """Analyzes text to produce an [Arousal, Valence] vector."""
        if not text:
            return np.array([0.1, 0.0]) # Default to neutral
        try:
            inputs = self.tokenizer(text, return_tensors="pt", truncation=True, padding=True).to(self.device)
            outputs = self.model(**inputs)
            scores = torch.softmax(outputs.logits, dim=1).cpu().numpy()[0]
            dominant_emotion_idx = np.argmax(scores)
            dominant_emotion = self.model.config.id2label[dominant_emotion_idx]
            
            arousal, valence = self.emotion_to_va.get(dominant_emotion, [0.1, 0.0])
            logger.info(f"Analyzed emotion: {dominant_emotion} -> Arousal={arousal:.2f}, Valence={valence:.2f}")
            return np.array([arousal, valence])
        except Exception as e:
            logger.error(f"Error during emotion analysis for text '{text[:50]}...': {e}", exc_info=True)
            return np.array([0.1, 0.0])

    def is_pillar_interaction(self, arousal: float, valence: float) -> bool:
        """
        Determines if an interaction is emotionally significant enough
        to be considered a "Pillar Memory" for core identity.
        """
        # High intensity is defined as high arousal and strong emotion (positive or negative)
        return arousal > 0.75 and abs(valence) > 0.6

import torch
import torch.nn as nn

class DampedLinOSS(nn.Module):
    """Functional implementation of the Damped Linear Oscillatory State-Space model."""
    def __init__(self, state_dim: int = 16, input_dim: int = 4):
        super().__init__()
        self.state_dim = state_dim
        # Initialize weights to be small to promote stability at the start
        self.A = nn.Parameter(torch.randn(state_dim, state_dim) * 0.01)
        self.B = nn.Parameter(torch.randn(state_dim, input_dim) * 0.01)
        # Damping parameter, initialized to encourage some damping
        self.D = nn.Parameter(torch.randn(state_dim))

    def forward(self, state: torch.Tensor, input_features: torch.Tensor) -> torch.Tensor:
        """Implements the core mathematical update for the damped oscillator."""
        # Sigmoid ensures damping_factor is between 0 (no damping) and 1 (full damping)
        damping_factor = torch.sigmoid(self.D)

        internal_evolution = torch.matmul(state, self.A)
        external_input = torch.matmul(input_features, self.B.T)
        
        # Apply damping to the evolved state
        damped_evolution = internal_evolution * (1.0 - damping_factor)
        
        new_state = damped_evolution + external_input

        # Tanh keeps the state vector bounded between -1 and 1, preventing explosion
        return torch.tanh(new_state)

async def fetch_training_data(pool, user_id: str):
    """
    Fetches training data using a hybrid replay strategy.
    - All new Gold Star interactions.
    - All "Pillar" memories.
    - A random sample of older, "regular" Gold Star interactions.
    """
    async with pool.acquire() as conn:
        # 1. Fetch all new Gold Stars
        new_records = await conn.fetch("""
            SELECT i.transcript, i.aria_response
            FROM gold_star_data g JOIN interactions i ON g.interaction_id = i.interaction_id
            WHERE g.user_id = $1 AND g.is_processed = FALSE
        """, user_id)

        # 2. Fetch all Pillar memories
        pillar_records = await conn.fetch("""
            SELECT i.transcript, i.aria_response
            FROM gold_star_data g JOIN interactions i ON g.interaction_id = i.interaction_id
            WHERE g.user_id = $1 AND g.is_pillar_memory = TRUE
        """, user_id)

        # 3. Fetch a sample of old, regular Gold Stars
        regular_sample_records = await conn.fetch("""
            SELECT i.transcript, i.aria_response
            FROM gold_star_data g JOIN interactions i ON g.interaction_id = i.interaction_id
            WHERE g.user_id = $1 AND g.is_processed = TRUE AND g.is_pillar_memory = FALSE
            ORDER BY random() LIMIT 20
        """, user_id)

        # Mark new data as processed
        await conn.execute("UPDATE gold_star_data SET is_processed = TRUE WHERE user_id = $1 AND is_processed = FALSE", user_id)

        # Combine and deduplicate
        all_records = new_records + pillar_records + regular_sample_records
        unique_data = { (r['transcript'], r['aria_response']): r for r in all_records }.values()
        
        return [{"prompt": r['transcript'], "response": r['aria_response']} for r in unique_data]

@app.websocket("/ws/{user_id}")
async def websocket_endpoint(websocket: WebSocket, user_id: str, background_tasks: BackgroundTasks):
    await manager.connect(websocket, user_id)
    logger.info(f"User {user_id} connected.")
    
    try:
        user_state = await get_or_create_user_state(user_id)

        while True:
            audio_bytes = await websocket.receive_bytes()
            
            transcript, prosody_features = audio_processor.process_audio(audio_bytes)
            if not transcript:
                await manager.send_json({"type": "error", "message": "Could not understand audio."}, user_id)
                continue
            await manager.send_json({"type": "transcript", "data": transcript}, user_id)

            arousal, valence = analyst_core.analyze_arousal_valence(transcript)
            
            time_delta = time.time() - user_state.last_interaction_time
            interaction_features = np.array([arousal, valence, time_delta, prosody_features[0]])
            
            new_linoss_state_np = linoss_core.update_state(np.array(user_state.linoss_state), interaction_features)
            user_state.linoss_state = new_linoss_state_np.tolist()
            user_state.last_interaction_time = time.time()

            memories = memory_system.retrieve_causal_memories(user_id, transcript)
            
            aria_response = await agentic_core.generate_response(user_id, transcript, memories, user_state)
            
            async def store_and_respond(uid, trans, resp, aro, val):
                async with db_manager.db_pool.acquire() as conn:
                    interaction_id = await conn.fetchval(
                        "INSERT INTO interactions (user_id, transcript, aria_response, arousal, valence) VALUES ($1, $2, $3, $4, $5) RETURNING interaction_id;",
                        uid, trans, resp, float(aro), float(val)
                    )
                await manager.send_json({"type": "response", "data": resp, "interaction_id": str(interaction_id)}, uid)
                await memory_system.store_causal_memory(uid, interaction_id, trans, resp)
                await db_manager.redis_pool.set(f"user_state:{uid}", user_state.json())

            background_tasks.add_task(store_and_respond, user_id, transcript, aria_response, arousal, valence)

    except WebSocketDisconnect:
        logger.info(f"User {user_id} disconnected.")
    except Exception as e:
        logger.error(f"Error in WebSocket for user {user_id}: {e}", exc_info=True)
    finally:
        manager.disconnect(user_id)
