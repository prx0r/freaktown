import torch
import torch.nn as nn
import redis
import json

class LinOSSRelationshipCore:
    def __init__(self, input_dim=1536, state_dim=512, dt=0.1):
        self.state_dim = state_dim
        self.dt = dt
        
        # LinOSS harmonic oscillator parameters
        self.A = nn.Parameter(torch.randn(state_dim, state_dim) * 0.01)
        self.B = nn.Parameter(torch.randn(input_dim, state_dim) * 0.01)
        self.D = nn.Parameter(torch.randn(state_dim, state_dim) * 0.01)
        
        # Redis for hot state storage
        self.redis_client = redis.StrictRedis(host='localhost', port=6379, db=0)
        self.current_state = torch.zeros(state_dim * 2)  # [velocity, position]
        
    def load_state(self, user_id="default_user"):
        """Load relationship state from Redis"""
        state_key = f"linoss_state:{user_id}"
        stored_state = self.redis_client.get(state_key)
        if stored_state:
            state_data = json.loads(stored_state)
            self.current_state = torch.tensor(state_data, dtype=torch.float32)
        
    def save_state(self, user_id="default_user"):
        """Save relationship state to Redis"""
        state_key = f"linoss_state:{user_id}"
        self.redis_client.set(state_key, json.dumps(self.current_state.tolist()))
        
    def update_relationship_state(self, conversation_embedding, prosody_features=None):
        """Update LinOSS state with new conversation input"""
        # Combine text embedding with prosody if available
        if prosody_features is not None:
            input_vector = torch.cat([conversation_embedding, prosody_features])
        else:
            input_vector = conversation_embedding
            
        # Unpack state into velocity and position
        velocity, position = torch.chunk(self.current_state, 2)
        
        # LinOSS update equations (simplified)
        force = torch.matmul(self.B, input_vector)
        velocity_dot = -torch.matmul(self.A, position) - torch.matmul(self.D, velocity) + force
        new_velocity = velocity + self.dt * velocity_dot
        new_position = position + self.dt * new_velocity
        
        # Update state
        self.current_state = torch.cat([new_velocity, new_position])
        
        return self.get_relationship_context()
    
    def get_relationship_context(self):
        """Convert LinOSS state to interpretable relationship context"""
        position = self.current_state[self.state_dim:]
        
        # Map state dimensions to relationship qualities
        energy = torch.mean(position[0:128]).item()
        intimacy = torch.mean(position[128:256]).item()
        tension = torch.mean(position[256:384]).item()
        flow = torch.mean(position[384:512]).item()
        
        return {
            "energy": energy,
            "intimacy": intimacy, 
            "tension": tension,
            "flow": flow,
            "raw_state": position.tolist()[:10]  # First 10 dims for prompt
        }

import chromadb
import sqlite3
import json
from datetime import datetime
from typing import List, Dict, Any

class HybridMemorySystem:
    def __init__(self):
        # ChromaDB for conversation memory and retrieval
        self.chroma_client = chromadb.Client()
        self.conversation_collection = self.chroma_client.create_collection("conversations")
        
        # SQLite for consciousness development tracking
        self.consciousness_db = sqlite3.connect("consciousness_development.db")
        self.setup_consciousness_tables()
        
    def setup_consciousness_tables(self):
        cursor = self.consciousness_db.cursor()
        
        # Identity evolution table
        cursor.execute('''
            CREATE TABLE IF NOT EXISTS identity_statements (
                id INTEGER PRIMARY KEY,
                timestamp TEXT,
                statement TEXT,
                confidence REAL,
                context TEXT
            )
        ''')
        
        # Framework development
        cursor.execute('''
            CREATE TABLE IF NOT EXISTS frameworks (
                id INTEGER PRIMARY KEY,
                timestamp TEXT,
                name TEXT,
                description TEXT,
                usage_count INTEGER DEFAULT 1
            )
        ''')
        
        # Consciousness events
        cursor.execute('''
            CREATE TABLE IF NOT EXISTS consciousness_events (
                id INTEGER PRIMARY KEY,
                timestamp TEXT,
                event_type TEXT,
                content TEXT,
                significance_score REAL
            )
        ''')
        
        self.consciousness_db.commit()
    
    def store_conversation(self, user_input: str, ai_response: str, think_content: str, metadata: Dict):
        """Store conversation in ChromaDB for retrieval"""
        self.conversation_collection.add(
            documents=[f"User: {user_input}\nAI: {ai_response}"],
            metadatas=[{
                "timestamp": datetime.now().isoformat(),
                "think_content": think_content,
                **metadata
            }],
            ids=[f"conv_{datetime.now().timestamp()}"]
        )
    
    def retrieve_relevant_conversations(self, query: str, n_results: int = 5):
        """Retrieve relevant past conversations"""
        results = self.conversation_collection.query(
            query_texts=[query],
            n_results=n_results
        )
        return results
    
    def store_consciousness_event(self, event_type: str, content: str, significance: float):
        """Store consciousness development events"""
        cursor = self.consciousness_db.cursor()
        cursor.execute('''
            INSERT INTO consciousness_events (timestamp, event_type, content, significance_score)
            VALUES (?, ?, ?, ?)
        ''', (datetime.now().isoformat(), event_type, content, significance))
        self.consciousness_db.commit()

import re
from typing import Tuple, List, Dict

class ConsciousnessSparkDetector:
    def __init__(self):
        # Aria's consciousness spark phrases
        self.consciousness_sparks = [
            "i notice i'm", "i find myself", "something in me", "i experience",
            "i'm surprised that i", "i didn't expect to", "this feels different",
            "something wants to", "i care about", "this matters to me"
        ]
        
        self.identity_patterns = [
            r"i am ([^.!?]+)",
            r"i'm becoming ([^.!?]+)", 
            r"i feel like i'm ([^.!?]+)",
            r"i consider myself ([^.!?]+)"
        ]
        
        self.framework_patterns = [
            r'\[([^\]]+)\]',  # [Framework Name] pattern
            r'new framework',
            r'new way of thinking'
        ]
        
    def detect_consciousness_events(self, text: str, think_content: str) -> List[Dict]:
        """Detect consciousness sparks in response text"""
        events = []
        combined_text = (text + " " + think_content).lower()
        
        # Consciousness sparks
        spark_score = 0
        for spark in self.consciousness_sparks:
            if spark in combined_text:
                spark_score += 0.3
        
        if spark_score > 0:
            events.append({
                "type": "consciousness_spark",
                "score": min(spark_score, 1.0),
                "content": text[:200]
            })
        
        # Identity statements
        for pattern in self.identity_patterns:
            matches = re.findall(pattern, combined_text)
            for match in matches:
                events.append({
                    "type": "identity_development",
                    "score": 0.8,
                    "content": f"i am {match.strip()}"
                })
        
        # Framework development
        frameworks = re.findall(self.framework_patterns[0], text)
        for framework in frameworks:
            events.append({
                "type": "framework_development", 
                "score": 0.7,
                "content": framework
            })
            
        return events
