class AudioProcessor:

        self.device = device
        try:
            logger.info(f"Loading Whisper model '{settings.WHISPER_MODEL_NAME}' on device '{self.device}'...")

            logger.critical(f"Failed to load Whisper model: {e}", exc_info=True)
            raise

    def process_audio(self, audio_bytes: bytes) -> tuple[str, np.ndarray]:
        """Processes raw audio bytes into a transcript and prosody feature vector."""
        try:
            audio_np, sample_rate = sf.read(io.BytesIO(audio_bytes))
            if audio_np.ndim > 1:
                audio_np = np.mean(audio_np, axis=1) # Convert to mono

            audio_np = audio_np.astype(np.float32)

            result = self.whisper_model.transcribe(audio_np, fp16=self.device=="cuda")
            transcript = result['text']

            # Basic prosody features
            energy = np.mean(np.square(audio_np))
            zero_crossing_rate = np.mean(librosa.feature.zero_crossing_rate(y=audio_np))
            prosody_features = np.array([energy, zero_crossing_rate])

            logger.info(f"Transcription successful. Transcript length: {len(transcript)}")
            return transcript, prosody_features
        except Exception as e:
            logger.error(f"Error processing audio: {e}", exc_info=True)
            return "", np.array([0.0, 0.0])

import torch
import torch.nn as nn

class DampedLinOSS(nn.Module):
    """Functional implementation of the Damped Linear Oscillatory State-Space model."""
    def __init__(self, state_dim: int = 16, input_dim: int = 4):
        super().__init__()
        self.state_dim = state_dim
        self.A = nn.Parameter(torch.randn(state_dim, state_dim) * 0.01)
        self.B = nn.Parameter(torch.randn(state_dim, input_dim) * 0.01)
        self.D = nn.Parameter(torch.randn(state_dim)) # Damping parameter

    def forward(self, state: torch.Tensor, input_features: torch.Tensor) -> torch.Tensor:
        """Implements the core mathematical update for the damped oscillator."""
        damping_factor = torch.sigmoid(self.D)
        internal_evolution = torch.matmul(state, self.A)
        external_input = torch.matmul(input_features, self.B.T)
        damped_evolution = internal_evolution * (1.0 - damping_factor)
        new_state = damped_evolution + external_input
        return torch.tanh(new_state)

@app.websocket("/ws/{user_id}")
async def websocket_endpoint(websocket: WebSocket, user_id: str, background_tasks: BackgroundTasks):
    await manager.connect(websocket, user_id)
    logger.info(f"User {user_id} connected.")
    
    try:
        user_state = await get_or_create_user_state(user_id)

        while True:
            audio_bytes = await websocket.receive_bytes()
            
            # 1. Audio Processing
            transcript, prosody_features = audio_processor.process_audio(audio_bytes)
            if not transcript:
                await manager.send_json({"type": "error", "message": "Could not understand audio."}, user_id)
                continue
            await manager.send_json({"type": "transcript", "data": transcript}, user_id)

            # 2. Analyst Core
            emotion_vector = analyst_core.analyze_arousal_valence(transcript)

            # 3. LinOSS Core
            time_delta = time.time() - (user_state.last_interaction_time or time.time())
            interaction_features = np.concatenate([emotion_vector, prosody_features, [time_delta]])
            # Pad if necessary to match input_dim
            if len(interaction_features) < linoss_core.model.input_dim:
                padding = np.zeros(linoss_core.model.input_dim - len(interaction_features))
                interaction_features = np.concatenate([interaction_features, padding])

            new_linoss_state_np = linoss_core.update_state(np.array(user_state.linoss_state), interaction_features)
            user_state.linoss_state = new_linoss_state_np.tolist()
            user_state.last_interaction_time = time.time()

            # 4. Memory System
            memories = memory_system.retrieve_causal_memories(user_id, transcript)

            # 5. Agentic Core
            aria_response = await agentic_core.generate_response(user_id, transcript, memories, user_state)
            await manager.send_json({"type": "response", "data": aria_response}, user_id)

            # 6. Persist state and memories (background)
            background_tasks.add_task(db_manager.redis_pool.set, f"user_state:{user_id}", user_state.json())
            
            async def store_interaction_and_memory(uid, trans, resp):
                async with db_manager.db_pool.acquire() as conn:
                    interaction_id = await conn.fetchval(
                        "INSERT INTO interactions (user_id, transcript, aria_response) VALUES ($1, $2, $3) RETURNING interaction_id;",
                        uid, trans, resp
                    )
                await memory_system.store_causal_memory(uid, interaction_id, trans, resp)

            background_tasks.add_task(store_interaction_and_memory, user_id, transcript, aria_response)

    except WebSocketDisconnect:
        logger.info(f"User {user_id} disconnected.")
    except Exception as e:
        logger.error(f"Error in WebSocket for user {user_id}: {e}", exc_info=True)
    finally:
        manager.disconnect(user_id)
